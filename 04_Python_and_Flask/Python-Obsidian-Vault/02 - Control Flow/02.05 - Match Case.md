## Basic Match Case

```python
def get_status_message(http_status: int) -> str:
    """Returns a message based on the HTTP status code."""
    match http_status:
        case 200:
            return "OK"
        case 404:
            return "Not Found"
        case 500:
            return "Server Error"
        case _:
            return "Unknown status"

print(get_status_message(200))
```

## Matching Sequences

```python
from typing import Any

def process_command(command: list[Any]) -> None:
    """Processes commands using sequence patterns."""
    match command:
        case ["quit"]:
            print("Quitting...")
        case ["move", int(x), int(y)]:
            print(f"Moving to ({x}, {y})")
        case ["circle", float(radius)]:
            print(f"Drawing circle with radius {radius}")
        case _:
            print("Unknown command format")

process_command(["move", 10, 20])
```

## Matching Dictionaries

```python
def process_user(user: dict[str, Any]) -> None:
    """Matches dictionary structures."""
    match user:
        case {"name": str(name), "role": "admin"}:
            print(f"Admin Access: {name}")
        case {"name": str(name), "role": "user"}:
            print(f"User Access: {name}")
        case {"name": str(name)}:
            print(f"Guest Access: {name}")

process_user({"name": "Alice", "role": "admin"})
```

## Guards (Conditions)

```python
def classify_age(age: int) -> str:
    """Classifies age using match guards."""
    match age:
        case n if n < 0:
            return "Invalid age"
        case n if n < 13:
            return "Child"
        case n if n < 20:
            return "Teenager"
        case n if n < 65:
            return "Adult"
        case _:
            return "Senior"

print(classify_age(25))
```

## 🚀 Advanced Topics

### Class Patterns and Custom Matching
Python 3.12+ makes pattern matching exceptionally powerful for data classes and custom objects.

```python
from dataclasses import dataclass

@dataclass
class Point:
    x: float
    y: float

def describe_point(p: Point) -> str:
    """Matches attributes of an object."""
    match p:
        case Point(x=0.0, y=0.0):
            return "Origin"
        case Point(x=0.0, y=y):
            return f"On Y-axis at {y}"
        case Point(x=x, y=0.0):
            return f"On X-axis at {x}"
        case Point():
            return "Somewhere in space"
        case _:
            return "Not a point"

print(describe_point(Point(0.0, 5.0)))
```

### Async Event Processing with Match
Pattern matching is heavily used in async systems to route incoming websocket or network events efficiently.

```python
import asyncio
from typing import Any

async def route_event(event: dict[str, Any]) -> None:
    """Routes network events based on their payload structure."""
    match event:
        case {"type": "ping", "timestamp": float(ts)}:
            print(f"Pong! at {ts}")
            await asyncio.sleep(0.1)
        case {"type": "data", "payload": list(data)}:
            print(f"Processing data array: {data}")
            await asyncio.sleep(0.2)
        case _:
            print("Unknown event")

asyncio.run(route_event({"type": "ping", "timestamp": 1234.5}))
```

### Memory and GIL Considerations
`match-case` is highly optimized in CPython. Unlike a long chain of `if-elif` statements which evaluate sequentially in Python bytecodes (and thus hold the GIL longer for complex evaluations), dictionary and class pattern matches are resolved at the C level rapidly. In a free-threaded PEP 703 world, structural matching over immutable data (like tuples) ensures thread-safe routing without locks.
