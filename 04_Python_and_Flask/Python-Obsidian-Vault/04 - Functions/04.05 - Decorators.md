## Basic Decorator

```python
from typing import Callable, Any
from functools import wraps

def my_decorator(func: Callable[..., Any]) -> Callable[..., Any]:
    """
    A basic decorator that prints before and after function execution.
    Preserves original function metadata using @wraps.
    """
    @wraps(func)
    def wrapper(*args: Any, **kwargs: Any) -> Any:
        print(f"Before calling {func.__name__}")
        result = func(*args, **kwargs)
        print(f"After calling {func.__name__}")
        return result
    return wrapper

@my_decorator
def say_hello() -> None:
    """Prints a simple greeting."""
    print("Hello!")

if __name__ == "__main__":
    say_hello()
```

## Decorator with Arguments

```python
from typing import Callable, Any
from functools import wraps

def repeat(num_times: int) -> Callable[[Callable[..., Any]], Callable[..., Any]]:
    """
    Decorator that repeats the execution of a function `num_times`.
    """
    def decorator(func: Callable[..., Any]) -> Callable[..., Any]:
        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            result = None
            for _ in range(num_times):
                result = func(*args, **kwargs)
            return result
        return wrapper
    return decorator

@repeat(num_times=3)
def greet(name: str) -> None:
    """Greets the provided name."""
    print(f"Hello, {name}!")

if __name__ == "__main__":
    greet("Alice")
```

## Common Built-in Decorators (`functools` Deep Dive)

The `functools` module provides highly optimized, C-implemented decorators for common use cases.

```python
import time
from functools import lru_cache, cache
from typing import Callable, Any

def timer(func: Callable[..., Any]) -> Callable[..., Any]:
    """Measures execution time of a function."""
    @wraps(func)
    def wrapper(*args: Any, **kwargs: Any) -> Any:
        start: float = time.perf_counter()
        result: Any = func(*args, **kwargs)
        print(f"{func.__name__} took {time.perf_counter() - start:.4f}s")
        return result
    return wrapper

# Caching with lru_cache (memoization)
@lru_cache(maxsize=128)
def fibonacci(n: int) -> int:
    """Returns the nth Fibonacci number, utilizing an LRU cache."""
    if n < 2:
        return n
    return fibonacci(n-1) + fibonacci(n-2)

# Python 3.9+ introduced @cache (equivalent to lru_cache(maxsize=None))
@cache
def expensive_computation(x: int) -> int:
    time.sleep(0.1)  # Simulate work
    return x * x
```

## Asynchronous Decorators Deep Dive

With Python's `asyncio`, decorators must handle coroutines properly.

```python
import asyncio
from typing import Callable, Coroutine, Any
from functools import wraps
import time

def async_timer(func: Callable[..., Coroutine[Any, Any, Any]]) -> Callable[..., Coroutine[Any, Any, Any]]:
    """A timer decorator for async functions."""
    @wraps(func)
    async def wrapper(*args: Any, **kwargs: Any) -> Any:
        start_time = time.perf_counter()
        # await the execution of the coroutine
        result = await func(*args, **kwargs)
        end_time = time.perf_counter()
        print(f"Async {func.__name__} took {end_time - start_time:.4f}s")
        return result
    return wrapper

@async_timer
async def fetch_data(delay: float) -> dict[str, str]:
    """Simulates fetching data asynchronously."""
    await asyncio.sleep(delay)
    return {"status": "success", "data": "Sample content"}

async def main() -> None:
    data = await fetch_data(1.5)
    print(data)

if __name__ == "__main__":
    asyncio.run(main())
```

## 🧪 Practice

### Exercise 1: Logging Decorator
```python
from typing import Callable, Any
import logging

logging.basicConfig(level=logging.INFO)

def log_call(func: Callable[..., Any]) -> Callable[..., Any]:
    """Logs the arguments and return value of a function."""
    @wraps(func)
    def wrapper(*args: Any, **kwargs: Any) -> Any:
        logging.info(f"Calling {func.__name__} with args={args}, kwargs={kwargs}")
        result = func(*args, **kwargs)
        logging.info(f"{func.__name__} returned {result}")
        return result
    return wrapper

@log_call
def add(a: int | float, b: int | float) -> int | float:
    return a + b

if __name__ == "__main__":
    add(3, 5.5)
```

### Exercise 2: Authorization Decorator
```python
from typing import Callable, Any

def require_login(func: Callable[..., Any]) -> Callable[..., Any]:
    """Checks if a user dictionary contains an 'is_logged_in' flag."""
    @wraps(func)
    def wrapper(user: dict[str, Any], *args: Any, **kwargs: Any) -> Any:
        if not user.get("is_logged_in"):
            return "Please log in first"
        return func(user, *args, **kwargs)
    return wrapper

@require_login
def view_profile(user: dict[str, Any]) -> str:
    return f"Profile: {user.get('name', 'Unknown')}"

if __name__ == "__main__":
    print(view_profile({"name": "Alice", "is_logged_in": True}))
    print(view_profile({"name": "Bob", "is_logged_in": False}))
```
