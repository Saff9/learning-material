## Raising Built-in Exceptions

```python
def divide(a: int | float, b: int | float) -> float:
    """Divides two numbers, raising ZeroDivisionError explicitly if needed."""
    if b == 0:
        raise ZeroDivisionError("Cannot divide by zero")
    return a / b

def parse_age(age_string: str) -> int:
    """Parses age, raising ValueError on invalid states."""
    age: int = int(age_string)
    if age < 0:
        raise ValueError("Age cannot be negative")
    return age
```

## Re-raising Exceptions
You can catch an exception, log it or modify state, and then use `raise` without arguments to throw it upwards.

```python
def process_file(filename: str) -> str:
    """Reads a file, logging and re-raising FileNotFoundError."""
    try:
        with open(filename, "r") as f:
            return f.read()
    except FileNotFoundError:
        print(f"Warning: Failed attempt to open {filename}")
        raise  # Re-raise the exact same exception with original traceback
```

## Exception Chaining (`raise ... from`)
When wrapping exceptions, you should use `from` to link the new exception to the original cause. This produces a clear traceback.

```python
import json
from typing import Any

class ConfigError(Exception):
    pass

def read_config(filename: str) -> dict[str, Any]:
    """Reads JSON config with chained exceptions."""
    try:
        with open(filename, "r") as f:
            return json.load(f)
    except FileNotFoundError as e:
        # e.__cause__ will be set to the FileNotFoundError
        raise ConfigError(f"Config file missing: {filename}") from e
    except json.JSONDecodeError as e:
        raise ConfigError(f"Invalid JSON in {filename}") from e
```

## Suppressing Context (`from None`)
Sometimes the underlying exception is an implementation detail you want to hide from the caller.

```python
class CustomSecurityError(Exception):
    pass

def validate_token(token: str) -> bool:
    """Validates token, hiding internal ValueError exceptions."""
    try:
        parsed = int(token)
    except ValueError:
        # Hides the ValueError in the traceback completely
        raise CustomSecurityError("Invalid token format") from None
    return True
```

## Assertions
Assertions are for internal sanity checks and invariant testing. They should **never** be used for data validation, because Python can be run with `-O` (optimize), which strips out all assert statements.

```python
def calculate_discount(price: float, discount: float) -> float:
    """Calculates discount with internal invariant assertions."""
    # DO NOT use assertions for user input validation!
    assert price > 0, "Price must be positive"
    assert 0 <= discount <= 1, "Discount must be between 0 and 1"
    
    return price * (1 - discount)
```

### Exercise 1: Input Validator with Chaining
```python
def get_positive_number(prompt: str) -> float:
    """Gets a positive number, demonstrating exception chaining."""
    value: str = input(prompt)
    try:
        num: float = float(value)
        if num <= 0:
            raise ValueError("Number must be positive")
        return num
    except ValueError as e:
        # Chain the ValueError we just caught (if float conversion failed) 
        # or the one we just raised (if num <= 0)
        raise ValueError(f"Invalid input: '{value}'. Detail: {e}") from e
```

### Exercise 2: API Error Handler
```python
from typing import Any

class APIError(Exception): pass
class NotFoundError(APIError): pass
class UnauthorizedError(APIError): pass

def api_call(endpoint: str) -> list[Any] | dict[str, Any]:
    """Simulates an API call with structured error raising."""
    responses: dict[str, dict[str, Any]] = {
        "/users": {"status": 200, "data": [{"id": 1}]},
        "/admin": {"status": 401, "error": "Unauthorized"},
        "/missing": {"status": 404, "error": "Not found"}
    }

    response = responses.get(endpoint, {"status": 500, "error": "Internal Error"})

    match response["status"]:
        case 404:
            raise NotFoundError(f"Endpoint {endpoint} not found")
        case 401:
            raise UnauthorizedError("Authentication required")
        case 200:
            return response.get("data", [])
        case _:
            raise APIError(f"API error: {response}")

try:
    data = api_call("/admin")
except UnauthorizedError as e:
    print(f"Access denied: {e}")
```

4. What is the difference between `raise` and `assert`?
5. Why is using `assert` for user data validation dangerous?

**Answers:**
1. Use `raise` without any arguments.
2. Linking a new exception to the one that caused it using `from e`, preserving the traceback of both.
3. When you want to hide the original exception context (e.g. hiding internal implementation details).
4. `raise` guarantees an error is thrown. `assert` is an internal check for developers.
5. Because running Python with the `-O` flag removes all assertions, leaving your application without validation.
