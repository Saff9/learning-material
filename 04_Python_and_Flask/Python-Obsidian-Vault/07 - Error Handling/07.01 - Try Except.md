## Basic Try-Except

```python
def safe_divide_basic() -> None:
    """Demonstrates basic try-except block."""
    try:
        result: float = 10 / 0
    except ZeroDivisionError:
        print("Cannot divide by zero!")

safe_divide_basic()
```

## Catching Specific Exceptions

```python
def process_user_input() -> None:
    """Handles specific exceptions from user input."""
    try:
        number_str: str = input("Enter a number: ")
        number: int = int(number_str)
        result: float = 100 / number
    except ValueError:
        print("That is not a valid number!")
    except ZeroDivisionError:
        print("Cannot divide by zero!")

# process_user_input()
```

## Catching Multiple Exceptions

```python
def read_and_divide() -> None:
    """Handles multiple exceptions in a single block or multiple blocks."""
    try:
        with open("data.txt", "r") as file:
            number: int = int(file.read())
            result: float = 100 / number
    except (ValueError, ZeroDivisionError) as e:
        print(f"Invalid data or division by zero: {e}")
    except FileNotFoundError:
        print("File not found!")

read_and_divide()
```

## Exception Groups and `except*` (Python 3.11+)
Python 3.11 introduced `ExceptionGroup` to raise and handle multiple unrelated exceptions simultaneously, heavily used in modern async code (`asyncio.TaskGroup`).

```python
def handle_exception_group() -> None:
    """Demonstrates Python 3.11+ ExceptionGroup and except* syntax."""
    try:
        raise ExceptionGroup(
            "Multiple errors occurred",
            [ValueError("Invalid value"), TypeError("Type mismatch")]
        )
    except* ValueError as e:
        print(f"Handled ValueErrors: {e.exceptions}")
    except* TypeError as e:
        print(f"Handled TypeErrors: {e.exceptions}")

handle_exception_group()
```

## Catching Any Exception

```python
def risky_operation() -> None:
    """A risky operation that might fail."""
    raise RuntimeError("Something went horribly wrong")

def catch_all() -> None:
    """Catches general exceptions and inspects them."""
    try:
        risky_operation()
    except Exception as e:
        print(f"An error occurred: {e}")
        print(f"Error type: {type(e).__name__}")
        print(f"Error args: {e.args}")

catch_all()
```

### Exercise 1: Safe Division
```python
def safe_divide(a: int | float, b: int | float) -> float | str:
    """Safely divides two numbers, handling common arithmetic errors."""
    try:
        return a / b
    except ZeroDivisionError:
        return "Cannot divide by zero"
    except TypeError:
        return "Invalid input types"

print(safe_divide(10, 2))   # 5.0
print(safe_divide(10, 0))   # Cannot divide by zero
print(safe_divide(10, "a")) # Invalid input types
```

### Exercise 2: Safe Async Execution
Using modern Python features to handle errors asynchronously.

```python
import asyncio

async def fetch_data(id: int) -> dict[str, str]:
    """Simulates fetching data asynchronously."""
    if id < 0:
        raise ValueError(f"Invalid ID: {id}")
    await asyncio.sleep(0.1)
    return {"id": str(id), "data": "Sample"}

async def main() -> None:
    """Main async runner demonstrating TaskGroup (Python 3.11+)."""
    try:
        async with asyncio.TaskGroup() as tg:
            task1 = tg.create_task(fetch_data(1))
            task2 = tg.create_task(fetch_data(-1)) # Will fail
    except* ValueError as eg:
        print("Caught validation errors in tasks!")
        for error in eg.exceptions:
            print(f"- {error}")

if __name__ == "__main__":
    asyncio.run(main())
```

1. What is the purpose of try-except?
2. Should you catch `Exception` or specific exceptions?
3. What is `ExceptionGroup` and when is it useful?
4. What happens if an exception is not caught?
5. What is the base class for all exceptions?

**Answers:**
1. To handle errors gracefully without crashing the application.
2. Specific exceptions - this is better practice to avoid masking unrelated bugs.

5. `BaseException` (but `Exception` is used as the base for regular application errors).
