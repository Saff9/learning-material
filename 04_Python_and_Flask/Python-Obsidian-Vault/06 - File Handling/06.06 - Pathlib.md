## Creating and Joining Paths

```python
from pathlib import Path

def demonstrate_paths() -> None:
    """Shows how to create and manipulate path objects."""
    # Current directory
    current_dir: Path = Path(".")

    # Absolute path
    home: Path = Path.home()
    
    # Joining paths using the '/' operator (Overloaded __truediv__)
    documents: Path = home / "Documents"
    
    # Building paths
    base = Path("/home/user")
    file_path = base / "documents" / "file.txt"
    print(file_path)  # /home/user/documents/file.txt
```

## Path Properties

```python
def inspect_path(p: Path) -> None:
    """Prints various properties of a Path object."""
    print(p.name)       # file.txt (Full filename)
    print(p.stem)       # file (Filename without extension)
    print(p.suffix)     # .txt (Extension)
    print(p.parent)     # Directory containing the file
    print(p.parents[0]) # Immediate parent
    print(p.parts)      # Tuple of path components
    print(p.absolute()) # Absolute, resolved path
```

## Checking Files

```python
def check_path_status(p: Path) -> None:
    """Checks the existence and type of a path."""
    print(p.exists())      # True if it exists
    print(p.is_file())     # True if it is a regular file
    print(p.is_dir())      # True if it is a directory
    print(p.is_absolute()) # True if it's an absolute path
```

## Direct File Operations

`Path` objects offer convenience methods to read and write entire files directly, avoiding explicit `open()` calls.

```python
def read_write_directly(p: Path) -> None:
    """Reads and writes data directly using pathlib."""
    # Write
    p.write_text("Hello, World!", encoding="utf-8")
    p.write_bytes(b"\\x00\\x01\\x02")

    # Read
    content: str = p.read_text(encoding="utf-8")
    bytes_data: bytes = p.read_bytes()

    # Rename
    new_path = p.with_name("new_name.txt")
    p.rename(new_path)

    # Delete
    if p.exists():
        p.unlink()  # Remove file
```

## Directory Operations and Globbing

```python
from typing import Iterator

def directory_operations(target_dir: Path) -> None:
    """Performs iteration and globbing on directories."""
    # Create directory (including parents if missing)
    target_dir.mkdir(parents=True, exist_ok=True)
    
    # List immediate children
    for item in target_dir.iterdir():
        print(item)

    # Search for pattern in immediate directory
    py_files: Iterator[Path] = target_dir.glob("*.py")
    
    # Recursive search
    all_txt_files: Iterator[Path] = target_dir.rglob("*.txt")
```

### Exercise 1: File Organizer
```python
from pathlib import Path
import shutil

def organize_by_extension(directory: str) -> None:
    """Organizes files in a directory into subfolders by their extensions."""
    path = Path(directory)
    for file in path.iterdir():
        if file.is_file():
            ext = file.suffix[1:] or "no_extension"
            folder = path / ext
            folder.mkdir(exist_ok=True)
            # Move file using shutil
            shutil.move(str(file), str(folder / file.name))
```

### Exercise 2: Find Large Files
```python
from pathlib import Path

def find_large_files(directory: str, min_size_mb: float = 10.0) -> None:
    """Recursively finds files larger than a specified threshold."""
    path = Path(directory)
    min_bytes: float = min_size_mb * 1024 * 1024

    for file in path.rglob("*"):
        if file.is_file():
            size = file.stat().st_size
            if size > min_bytes:
                size_mb = size / (1024 * 1024)
                print(f"{file}: {size_mb:.1f} MB")
```

### Exercise 3: Backup Script
```python
from pathlib import Path
import shutil
from datetime import datetime

def backup_directory(source: str, backup_location: str) -> None:
    """Creates a timestamped backup of a directory."""
    source_path = Path(source)
    timestamp: str = datetime.now().strftime("%Y%m%d_%H%M%S")
    backup_name: str = f"{source_path.name}_{timestamp}"
    backup_path = Path(backup_location) / backup_name

    shutil.copytree(source_path, backup_path)
    print(f"Backup created: {backup_path}")
```
