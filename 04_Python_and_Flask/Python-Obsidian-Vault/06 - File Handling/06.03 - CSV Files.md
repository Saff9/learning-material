## Reading CSV Files

```python
import csv
from typing import Any, Iterator

def read_basic_csv(filepath: str) -> list[list[str]]:
    """
    Reads a simple CSV file where each row is a list of strings.
    
    Args:
        filepath (str): Path to CSV file.
        
    Returns:
        list[list[str]]: A list of rows.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        reader = csv.reader(f)
        return list(reader)
```

## Reading with Headers (DictReader)

```python
import csv
from typing import Iterator

def read_csv_dict(filepath: str) -> Iterator[dict[str, str]]:
    """
    Yields rows as dictionaries, using the first row as headers.
    Memory efficient for large datasets.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        for row in reader:
            yield row
```

## Writing CSV Files

```python
import csv

def write_basic_csv(filepath: str, data: list[list[str]]) -> None:
    """
    Writes a list of rows to a CSV file.
    
    Args:
        filepath (str): Destination path.
        data (list[list[str]]): Data to write.
    """
    # newline="" is required for csv module to prevent double-spacing on Windows
    with open(filepath, "w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerows(data)
```

## DictWriter

```python
import csv

def write_dict_csv(filepath: str, data: list[dict[str, Any]], fieldnames: list[str]) -> None:
    """Writes a list of dictionaries to a CSV file."""
    with open(filepath, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(data)
```

## Advanced Processing with itertools

```python
import csv
import itertools
from typing import Iterator

def process_large_csv_in_chunks(filepath: str, chunk_size: int = 1000) -> Iterator[list[dict[str, str]]]:
    """
    Reads a large CSV in chunks using itertools.islice for memory efficiency.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        while True:
            chunk: list[dict[str, str]] = list(itertools.islice(reader, chunk_size))
            if not chunk:
                break
            yield chunk
```

### Exercise 1: CSV to Dictionary

import csv

def csv_to_dict(filename: str) -> list[dict[str, str]]:
    """Converts a CSV file to a list of dictionaries."""
    with open(filename, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)

### Exercise 2: Filter CSV
```python
import csv

def filter_csv(input_file: str, output_file: str, min_age: int) -> None:
    """Filters CSV rows based on an age condition."""
    with open(input_file, "r", encoding="utf-8") as infile, \\
         open(output_file, "w", newline="", encoding="utf-8") as outfile:
         
        reader = csv.DictReader(infile)
        if not reader.fieldnames:
            return
            
        writer = csv.DictWriter(outfile, fieldnames=reader.fieldnames)
        writer.writeheader()

        for row in reader:
            if int(row["age"]) >= min_age:
                writer.writerow(row)
```
