## Deep Dive: The GIL and PEP 703
In CPython, the **Global Interpreter Lock (GIL)** is a mutex that protects access to Python objects, preventing multiple threads from executing Python bytecodes at once. This means multithreading in Python is best suited for **I/O-bound** tasks (network requests, file I/O). 
> **PEP 703 (Making the Global Interpreter Lock Optional)**: Starting experimentally in Python 13, CPython can be built without the GIL (`--disable-gil`), allowing true multithreaded parallelism for CPU-bound tasks.

## ThreadPoolExecutor (Modern Threading)

Using `ThreadPoolExecutor` is the recommended way to manage threads, rather than manually working with `threading.Thread`.

```python
import time
from concurrent.futures import ThreadPoolExecutor, as_completed

def io_bound_task(name: str, delay: float) -> str:
    """
    Simulates an I/O bound task like downloading a file.
    """
    print(f"Task {name} starting...")
    time.sleep(delay)
    return f"Task {name} finished."

def main() -> None:
    tasks: dict[str, float] = {"A": 1.5, "B": 1.0, "C": 2.0}
    
    # Using a context manager ensures threads are cleaned up
    with ThreadPoolExecutor(max_workers=3) as executor:
        # Submit tasks
        futures = {executor.submit(io_bound_task, name, delay): name 
                  for name, delay in tasks.items()}
        
        # Process as they complete
        for future in as_completed(futures):
            result: str = future.result()
            print(result)

if __name__ == "__main__":
    main()
```

## Custom Context Manager for Thread Synchronization

When threads share memory, race conditions can occur. We use locks to prevent this. We can wrap a lock in a custom context manager for cleaner code.

```python
import threading
import time
from collections import deque
from typing import Any, Self

class LockedResource:
    """
    A custom context manager providing thread-safe access to a resource.
    """
    def __init__(self) -> None:
        self._lock: threading.Lock = threading.Lock()
        self.queue: deque[int] = deque()

    def __enter__(self) -> Self:
        self._lock.acquire()
        return self

    def __exit__(self, exc_type: Any, exc_val: Any, exc_tb: Any) -> None:
        self._lock.release()

def worker(resource: LockedResource, worker_id: int) -> None:
    """Worker thread that safely mutates a shared resource."""
    with resource as res:
        print(f"Worker {worker_id} modifying queue.")
        res.queue.append(worker_id)
        time.sleep(0.1)

def main() -> None:
    shared_resource = LockedResource()
    threads: list[threading.Thread] = []
    
    for i in range(5):
        t = threading.Thread(target=worker, args=(shared_resource, i))
        threads.append(t)
        t.start()
        
    for t in threads:
        t.join()
        
    print(f"Final queue state: {list(shared_resource.queue)}")

if __name__ == "__main__":
    main()
```

## Using Itertools in Thread Pools

We can use `itertools.islice` or `itertools.batched` (Python 3.12+) to chunk workloads efficiently across threads.

```python
import itertools
from concurrent.futures import ThreadPoolExecutor

def process_batch(batch: tuple[int, ...]) -> int:
    """Processes a chunk of data."""
    return sum(batch)

def main() -> None:
    data: range = range(100)
    
    # Python 3.12+ itertools.batched
    batches = itertools.batched(data, 10)
    
    with ThreadPoolExecutor() as executor:
        results: list[int] = list(executor.map(process_batch, batches))
        
    print(f"Batch results: {results}")
    print(f"Total: {sum(results)}")

if __name__ == "__main__":
    main()
```
