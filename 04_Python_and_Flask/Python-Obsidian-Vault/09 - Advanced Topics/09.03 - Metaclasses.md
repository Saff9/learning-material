## What are Metaclasses?

In Python, classes themselves are objects. A metaclass is the "class of a class". The default metaclass in Python is `type`.

```python
class MyClass:
    pass

print(type(MyClass))  # <class 'type'>
print(type(int))      # <class 'type'>
```

## Creating Classes with `type()`

You can create classes dynamically using `type(name, bases, namespace)`.

```python
from typing import Any

# type(name: str, bases: tuple, namespace: dict) -> type
DynamicClass: type = type("DynamicClass", (), {"x": 1})

obj: Any = DynamicClass()
print(obj.x)  # 1
```

## Custom Metaclass

A custom metaclass overrides `__new__` (class creation) or `__init__` (class initialization).

```python
from typing import Any

class Meta(type):
    """A metaclass that modifies the class namespace."""
    def __new__(mcs: type, name: str, bases: tuple[type, ...], namespace: dict[str, Any]) -> type:
        print(f"Creating class: {name}")
        namespace["created_by"] = "Meta"
        return super().__new__(mcs, name, bases, namespace)

class MyClass(metaclass=Meta):
    x: int = 10

print(MyClass.created_by)  # Meta
```

## Modern Alternative: `__init_subclass__`

In modern Python, `__init_subclass__` is often preferred over metaclasses for simple class customizations and hook registries.

```python
from typing import Any

class PluginBase:
    registry: dict[str, type] = {}

    def __init_subclass__(cls: type, **kwargs: Any) -> None:
        super().__init_subclass__(**kwargs)
        PluginBase.registry[cls.__name__] = cls

class AudioPlugin(PluginBase): pass
class VideoPlugin(PluginBase): pass

print(PluginBase.registry) 
# {'AudioPlugin': <class '__main__.AudioPlugin'>, 'VideoPlugin': <class '__main__.VideoPlugin'>}
```

## The Singleton Metaclass & Concurrency

When writing singletons, consider thread safety, especially in multi-threaded environments or under PEP 703 (No GIL).

```python
import threading
from typing import Any

class ThreadSafeSingleton(type):
    """A thread-safe singleton metaclass."""
    _instances: dict[type, Any] = {}
    _lock = threading.Lock()

    def __call__(cls: type, *args: Any, **kwargs: Any) -> Any:
        with cls._lock:
            if cls not in cls._instances:
                cls._instances[cls] = super().__call__(*args, **kwargs)
        return cls._instances[cls]

class Database(metaclass=ThreadSafeSingleton):
    def __init__(self, dsn: str) -> None:
        self.dsn: str = dsn

db1 = Database("postgresql://localhost")
db2 = Database("mysql://localhost")
print(db1 is db2)  # True
```

### Exercise 1: Attribute Validator Metaclass
```python
from typing import Any

class ValidatedMeta(type):
    """Metaclass that warns if methods lack docstrings."""
    def __new__(mcs: type, name: str, bases: tuple[type, ...], namespace: dict[str, Any]) -> type:
        for key, value in namespace.items():
            if callable(value) and not key.startswith("__") and not getattr(value, "__doc__"):
                print(f"Warning: {name}.{key} lacks a docstring.")
        return super().__new__(mcs, name, bases, namespace)

class Service(metaclass=ValidatedMeta):
    def fetch_data(self) -> None:
        """Fetches the data."""
        pass

    def process_data(self) -> None:
        pass  # Triggers a warning
```

### Exercise 2: Auto-Property using `__init_subclass__`
```python
from typing import Any, Callable

class AutoPropertyBase:
    """Automatically converts _prefixed attributes to properties upon subclassing."""
    def __init_subclass__(cls: type, **kwargs: Any) -> None:
        super().__init_subclass__(**kwargs)
        for attr_name in list(vars(cls)):
            if attr_name.startswith("_") and not attr_name.startswith("__"):
                public_name: str = attr_name[1:]
                
                # Use a closure to capture the correct attr_name
                def make_getter(name: str) -> Callable[[Any], Any]:
                    return lambda self: getattr(self, name)
                
                setattr(cls, public_name, property(make_getter(attr_name)))

class Person(AutoPropertyBase):
    _name: str = "Default"
    
    def __init__(self, name: str) -> None:
        self._name = name

p = Person("Alice")
print(p.name)  # Alice
```
