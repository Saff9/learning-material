```python
# calculator.py
"""
A modern, asynchronous command-line calculator demonstrating Python 3.12+ features,
strict typing, context managers, and advanced standard library usage.
"""

import asyncio
import functools
import itertools
import time
from collections.abc import Callable, Awaitable
from typing import Any, TypeAlias

# Type alias for our operation functions (Python 3.12 style TypeAlias)
Operation: TypeAlias = Callable[[float, float], float]
AsyncOperation: TypeAlias = Callable[[float, float], Awaitable[float]]

class CalculationTimer:
    """A context manager to time calculations.
    
    Useful for profiling how long our mathematical operations take.
    """
    def __enter__(self) -> "CalculationTimer":
        self.start: float = time.perf_counter()
        return self

    def __exit__(self, exc_type: Any, exc_val: Any, exc_tb: Any) -> None:
        self.end: float = time.perf_counter()
        print(f"[Timer] Calculation took {self.end - self.start:.6f} seconds")

@functools.lru_cache(maxsize=128)
def add(a: float, b: float) -> float:
    """Add two numbers. Uses lru_cache for memoization of repeated additions."""
    return a + b

def subtract(a: float, b: float) -> float:
    """Subtract b from a."""
    return a - b

def multiply(a: float, b: float) -> float:
    """Multiply two numbers."""
    return a * b

def divide(a: float, b: float) -> float:
    """Divide a by b. Raises ValueError on division by zero."""
    if b == 0:
        raise ValueError("Cannot divide by zero")
    return a / b

def power(a: float, b: float) -> float:
    """Raise a to the power of b."""
    return a ** b

async def async_simulate_remote_calc(a: float, b: float, op: Operation) -> float:
    """
    Simulate a remote, asynchronous calculation using async/await.
    In Python 3.12+, this runs concurrently. Under PEP 703 (No-GIL), 
    true multi-threading could be utilized instead for CPU-bound tasks.
    """
    print("Sending calculation to remote server...")
    await asyncio.sleep(1) # Simulate network delay
    return op(a, b)

async def get_number(prompt: str) -> float:
    """Asynchronously get a valid number from user (for non-blocking I/O)."""
    while True:
        try:
            # Using asyncio.to_thread for blocking input in async code
            result: str = await asyncio.to_thread(input, prompt)
            return float(result)
        except ValueError:
            print("Invalid input. Please enter a number.")

def display_menu() -> None:
    """Display the calculator menu using itertools."""
    menu_items: list[str] = [
        "Add", "Subtract", "Multiply", "Divide", "Power", 
        "Async Add (Simulated Remote)", "Quit"
    ]
    print("\n=== Python 3.12+ Advanced Calculator ===")
    
    # Using itertools to create numbered list dynamically
    for num, item in zip(itertools.count(1), menu_items):
        print(f"{num}. {item}")

async def main() -> None:
    """Main asynchronous calculator loop."""
    operations: dict[str, tuple[str, Operation]] = {
        "1": ("Add", add),
        "2": ("Subtract", subtract),
        "3": ("Multiply", multiply),
        "4": ("Divide", divide),
        "5": ("Power", power),
        "6": ("Async Add", add),
    }

    while True:
        display_menu()
        choice: str = await asyncio.to_thread(input, "\nSelect operation (1-7): ")
        choice = choice.strip()

        # Using Python 3.10+ Structural Pattern Matching
        match choice:
            case "7":
                print("Goodbye!")
                break
            case c if c in operations:
                name, func = operations[c]
                print(f"\n{name}")

                num1: float = await get_number("Enter first number: ")
                num2: float = await get_number("Enter second number: ")

                try:
                    # Utilizing our custom context manager
                    with CalculationTimer():
                        if c == "6":
                            # Await the simulated asynchronous operation
                            result: float = await async_simulate_remote_calc(num1, num2, add)
                        else:
                            # Standard synchronous call
                            result = func(num1, num2)
                        print(f"\nResult: {result}")
                except ValueError as e:
                    print(f"\nError: {e}")
                except Exception as e:
                    print(f"\nUnexpected error: {e}")
            case _:
                print("Invalid choice. Please try again.")

if __name__ == "__main__":
    asyncio.run(main())
```

## Deep Dives into Python 3.12+ and Modern Patterns

### 1. `async`/`await` and non-blocking I/O
The application's `main()` loop and I/O tasks are wrapped in `async def`. Standard library functions like `input` are blocking operations. By wrapping them in `await asyncio.to_thread(input, prompt)`, we ensure the asyncio event loop remains non-blocked, permitting background tasks to run concurrently. We simulated a remote calculation to show how concurrent operations handle lag seamlessly.

### 2. Context Managers (`__enter__` and `__exit__`)
Context managers, defined using the `with` statement, ensure correct setup and teardown of resources. We created `CalculationTimer`, which implements `__enter__` (to start a timer) and `__exit__` (to calculate the duration). This ensures that profiling code isolates the target block properly and safely executes no matter what.

### 3. `functools.lru_cache` and `itertools`
- **`functools.lru_cache`**: Added to the `add` function, this memoizes outputs. If you call `add(2, 2)` a thousand times, the calculation executes once, returning the cached value instantly thereafter. 
- **`itertools.count`**: In `display_menu`, we dynamically pair numbers `1, 2, ...` to the menu items list using `zip(itertools.count(1), menu_items)`. This prevents hardcoded numbering bugs.

### 4. PEP 703: Making the Global Interpreter Lock (GIL) Optional
Python 3.13 introduces the experimental No-GIL mode (PEP 703). In standard Python (with the GIL), CPU-bound code (like calculating primes or matrix multiplication) cannot execute simultaneously on multiple cores in a single process, heavily handicapping multi-threading.
In our `async_simulate_remote_calc` function, if we replaced `asyncio.sleep` with heavy CPU-bound math, under a No-GIL build, you could use `concurrent.futures.ThreadPoolExecutor` to offload mathematical tasks and achieve true parallel processing without the overhead of `multiprocessing`.

## Common Bugs and Fixes

| Bug | Cause | Fix |
|-----|-------|-----|
| Mypy errors on callables | Using `typing.Callable` improperly | Use `collections.abc.Callable` in Python 3.9+ |
| Async loops freezing | Calling `input()` blocks event loop | Use `asyncio.to_thread(input, prompt)` |
| Menu looping infinitely | Missing pattern matching fallback | Implement `case _:` catch-all |
