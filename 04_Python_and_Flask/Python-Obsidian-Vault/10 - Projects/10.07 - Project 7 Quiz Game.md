```python
# quiz_game.py
"""
An interactive async quiz game with multiple categories.
Demonstrates async/await, type hinting, and custom context managers.
"""

import asyncio
import json
import random
import time
from pathlib import Path
from typing import Dict, List, Optional, Tuple, TypedDict
from contextlib import asynccontextmanager
import itertools
import functools

class QuestionData(TypedDict):
    """Type definition for a single question."""
    question: str
    options: List[str]
    answer: int

QuestionsDict = Dict[str, List[QuestionData]]

@asynccontextmanager
async def load_quiz_data(filename: str | Path) -> 'asyncio.Queue[QuestionsDict]':
    """
    Custom async context manager to load quiz data safely.
    Yields an asyncio.Queue containing the loaded questions data.
    """
    default_questions: QuestionsDict = {
        "python": [
            {
                "question": "What is the output of 2 ** 3?",
                "options": ["6", "8", "9", "4"],
                "answer": 1
            },
            {
                "question": "Which function converts a string to integer?",
                "options": ["str()", "int()", "float()", "bool()"],
                "answer": 1
            },
            {
                "question": "What does `len()` do?",
                "options": ["Returns length", "Returns type", "Returns id", "Returns hash"],
                "answer": 0
            }
        ],
        "general": [
            {
                "question": "What is the capital of France?",
                "options": ["London", "Berlin", "Paris", "Madrid"],
                "answer": 2
            }
        ]
    }
    
    path: Path = Path(filename)
    try:
        # Simulating async I/O, realistically useful for large files or DB fetching
        await asyncio.sleep(0.1) 
        if path.exists():
            with open(path, 'r', encoding='utf-8') as f:
                data: QuestionsDict = json.load(f)
        else:
            data = default_questions
        
        queue: asyncio.Queue[QuestionsDict] = asyncio.Queue()
        await queue.put(data)
        yield queue
    finally:
        # Cleanup phase goes here
        pass

class QuizGame:
    """Manages the async quiz game state and execution."""
    
    def __init__(self, questions: QuestionsDict) -> None:
        self.questions: QuestionsDict = questions
        self.score: int = 0
        self.total_questions: int = 0
        self.streak: int = 0
        self.best_streak: int = 0

    @functools.lru_cache(maxsize=32)
    def calculate_points(self, elapsed: int, base_points: int = 10) -> int:
        """
        Calculate points using caching for performance on repeated time intervals.
        Demonstrates functools.lru_cache.
        """
        return max(base_points - elapsed, 1)

    async def get_answer_with_timeout(self, timeout: float = 10.0) -> Optional[str]:
        """Gets user input asynchronously with a strict timeout limit."""
        loop: asyncio.AbstractEventLoop = asyncio.get_running_loop()
        try:
            # Note: Input blocking is delegated to a separate thread via run_in_executor
            answer: str = await asyncio.wait_for(
                loop.run_in_executor(None, input, "\nYour answer (1-4, or 'q' to quit) [10s timeout]: "),
                timeout=timeout
            )
            return answer.strip()
        except asyncio.TimeoutError:
            print("\n⏰ Time's up!")
            return None

    async def ask_question(self, question_data: QuestionData) -> Optional[Tuple[int, float]]:
        """Ask a single question and wait for the user's answer."""
        print(f"\n{question_data['question']}")
        for i, option in enumerate(question_data["options"]):
            print(f"{i + 1}. {option}")

        start_time: float = time.time()

        while True:
            answer_str: Optional[str] = await self.get_answer_with_timeout()
            
            if answer_str is None:
                # Timed out, represented as -1 for incorrect
                return -1, 10.0
            
            if answer_str.lower() == 'q':
                return None

            try:
                answer: int = int(answer_str) - 1
                if 0 <= answer < len(question_data["options"]):
                    elapsed: float = time.time() - start_time
                    return answer, elapsed
                else:
                    print("Invalid choice. Enter 1-4.")
            except ValueError:
                print("Please enter a valid number.")

    async def run_quiz(self, category: Optional[str] = None, num_questions: int = 5) -> None:
        """Run the main loop of the quiz asynchronously."""
        print("\n=== Python Async Quiz Game ===")

        available: List[QuestionData] = []
        if category and category in self.questions:
            available = self.questions[category]
        else:
            # Deep Dive: itertools.chain.from_iterable flattens a list of lists efficiently
            available = list(itertools.chain.from_iterable(self.questions.values()))

        selected: List[QuestionData] = random.sample(available, min(num_questions, len(available)))

        for i, q in enumerate(selected, 1):
            print(f"\n--- Question {i} of {len(selected)} ---")
            print(f"Score: {self.score} | Streak: {self.streak}")

            result: Optional[Tuple[int, float]] = await self.ask_question(q)
            if result is None:
                print("Exiting game...")
                break

            answer, elapsed = result
            self.total_questions += 1

            if answer == q["answer"]:
                points: int = self.calculate_points(int(elapsed))
                self.score += points
                self.streak += 1
                self.best_streak = max(self.best_streak, self.streak)
                print(f"✓ Correct! +{points} points (Time: {elapsed:.1f}s)")
            else:
                self.streak = 0
                correct: str = q["options"][q["answer"]]
                print(f"✗ Wrong or Too Slow! The answer was: {correct}")

        self.show_results()

    def show_results(self) -> None:
        """Display the final outcome of the quiz."""
        print("\n=== Quiz Complete ===")
        print(f"Questions answered: {self.total_questions}")
        print(f"Final score: {self.score}")
        print(f"Best streak: {self.best_streak}")

        if self.total_questions > 0:
            accuracy: float = (self.score / (self.total_questions * 10)) * 100
            print(f"Accuracy: {accuracy:.1f}%")


async def main() -> None:
    """Main async entry point for the game."""
    # Using our custom async context manager
    async with load_quiz_data("questions.json") as data_queue:
        questions: QuestionsDict = await data_queue.get()
        
        game = QuizGame(questions)

        print("Categories:")
        print("1. Python")
        print("2. General Knowledge")
        print("3. Mixed")

        choice: str = input("\nSelect category (1-3): ").strip()
        categories: Dict[str, Optional[str]] = {"1": "python", "2": "general", "3": None}
        category: Optional[str] = categories.get(choice)

        await game.run_quiz(category)

if __name__ == "__main__":
    # The new standard for running asyncio programs
    asyncio.run(main())
```

## 🧠 Deep Dives

### 1. `asyncio` and `await` Implementation
In this refactored quiz game, we've replaced standard procedural input flows with `asyncio`. The function `get_answer_with_timeout` uses `loop.run_in_executor()` coupled with `asyncio.wait_for()` to implement strict, non-blocking time constraints on user answers. In Python 3.12+, utilizing async frameworks is heavily normalized not only for network I/O, but also to build responsive UI logic and timeout loops.

### 2. Custom Context Managers (`@asynccontextmanager`)
We use `contextlib.asynccontextmanager` to create a reusable logic block `load_quiz_data`. Context managers are crucial for resource management (allocating and de-allocating cleanly without manual `finally` block repetitions). By making it asynchronous, we can safely perform database queries, file reads, or sleep tasks before yielding the data flow. 

### 3. Itertools and Functools
* **`itertools.chain.from_iterable()`**: Rather than looping over the dictionary values and manually appending/extending `available`, `itertools.chain` links all lists together at the C-level, providing a massive speed boost and maintaining cleaner syntax.
* **`functools.lru_cache`**: Caches the result of `calculate_points`. If you answer in `3` seconds multiple times, the function won't recalculate the math; it retrieves it straight from memory. 

### 4. PEP 703: Making the Global Interpreter Lock (GIL) Optional
Notice how `input()` blocks standard execution, meaning we had to shove it into an executor (`run_in_executor`)? PEP 703 targets removing the GIL. In a future with a fully no-GIL Python, thread-based operations won't need to juggle execution locks, meaning purely multi-threaded approaches (instead of purely `asyncio` based ones) can truly compute concurrently and seamlessly.

### 5. Strict Typing (`TypedDict`)
We avoid bare dictionaries by utilizing `TypedDict` from the `typing` module. `QuestionData` informs the IDE exactly what keys a question should have (`question`, `options`, `answer`), preventing runtime key errors and enabling robust autocomplete.

## Common Bugs and Fixes

| Bug | Cause | Fix |
|-----|-------|-----|
| `RuntimeError: Event loop is closed` | Running `asyncio.run()` manually multiple times | Let Python handle it strictly in the `if __name__ == "__main__":` block |
| Index out of range | Wrong answer index input | Strict bound checking implemented on `answer` length |
| `TypeError` during data fetching | Implicit Dict without Types | Forced with `QuestionsDict` type |
