```python
# budget_tracker.py
"""A personal budget tracking application utilizing Python 3.12+ features."""

import csv
import os
import asyncio
from datetime import datetime
from collections import defaultdict
from dataclasses import dataclass, asdict
from typing import Literal, Dict, List, Any, AsyncGenerator, Self
import itertools
import functools
import time

TransactionType = Literal["income", "expense"]

@dataclass(slots=True)
class Transaction:
    """Represents a single financial transaction."""
    date: str
    description: str
    amount: float
    category: str
    type_: TransactionType

    def to_dict(self) -> Dict[str, Any]:
        """Convert transaction to dictionary."""
        return asdict(self)

class TimerContext:
    """A custom context manager to measure operation duration."""
    def __init__(self, operation_name: str) -> None:
        self.operation_name: str = operation_name
        self.start_time: float = 0.0

    def __enter__(self) -> Self:
        """Start the timer on enter."""
        self.start_time = time.perf_counter()
        return self

    def __exit__(self, exc_type: Any, exc_val: Any, exc_tb: Any) -> None:
        """Stop the timer and print the elapsed time."""
        elapsed = time.perf_counter() - self.start_time
        print(f"[{self.operation_name}] took {elapsed:.4f} seconds.")

class BudgetTracker:
    """Manages transactions and calculations asynchronously."""
    
    def __init__(self, filename: str = "budget.csv") -> None:
        self.filename: str = filename
        self.transactions: List[Transaction] = []

    async def load(self) -> None:
        """Asynchronously load transactions from CSV (simulated async I/O)."""
        if not os.path.exists(self.filename):
            return

        with TimerContext("Loading Transactions"):
            # In a real async app, you might use aiofiles.
            # Here we simulate async file reading with asyncio.sleep for the event loop
            with open(self.filename, 'r', newline='') as f:
                reader = csv.DictReader(f)
                for row in reader:
                    await asyncio.sleep(0)  # Yield control to the event loop
                    self.transactions.append(Transaction(
                        date=row["date"],
                        description=row["description"],
                        amount=float(row["amount"]),
                        category=row["category"],
                        type_=row["type_"] # type: ignore
                    ))

    async def save(self) -> None:
        """Asynchronously save transactions to CSV."""
        with TimerContext("Saving Transactions"):
            with open(self.filename, 'w', newline='') as f:
                fieldnames = ["date", "description", "amount", "category", "type_"]
                writer = csv.DictWriter(f, fieldnames=fieldnames)
                writer.writeheader()
                for t in self.transactions:
                    await asyncio.sleep(0) # Yield control
                    writer.writerow(t.to_dict())

    async def add_transaction(self, description: str, amount: float, category: str, type_: TransactionType) -> None:
        """Add a new transaction and save."""
        date: str = datetime.now().strftime("%Y-%m-%d")
        transaction = Transaction(date, description, amount, category, type_)
        self.transactions.append(transaction)
        await self.save()
        print(f"Added: {description} - ${amount:.2f}")

    @functools.cached_property
    def balance(self) -> float:
        """Calculate current balance using functools.reduce for demonstration."""
        income: float = functools.reduce(
            lambda acc, t: acc + t.amount if t.type_ == "income" else acc,
            self.transactions,
            0.0
        )
        expenses: float = functools.reduce(
            lambda acc, t: acc + t.amount if t.type_ == "expense" else acc,
            self.transactions,
            0.0
        )
        return income - expenses

    def get_summary(self) -> Dict[str, Dict[str, float]]:
        """Get spending summary by category using itertools.groupby."""
        # groupby requires the data to be sorted by the grouping key
        sorted_transactions = sorted(self.transactions, key=lambda t: t.category)
        summary: Dict[str, Dict[str, float]] = defaultdict(lambda: {"income": 0.0, "expense": 0.0})

        for category, group in itertools.groupby(sorted_transactions, key=lambda t: t.category):
            for t in group:
                summary[category][t.type_] += t.amount

        return dict(summary)

    def show_report(self) -> None:
        """Display financial report."""
        print("\n=== Budget Report ===")
        
        income = sum(t.amount for t in self.transactions if t.type_ == "income")
        expenses = sum(t.amount for t in self.transactions if t.type_ == "expense")
        # In Python 3.12, we can re-evaluate cached_property if needed, but here we just compute
        balance = income - expenses

        print(f"Total Income:  ${income:.2f}")
        print(f"Total Expenses: ${expenses:.2f}")
        print(f"Balance:        ${balance:.2f}")

        print("\n--- By Category ---")
        summary = self.get_summary()
        for category, amounts in sorted(summary.items()):
            net: float = amounts["income"] - amounts["expense"]
            print(f"{category:15} Income: ${amounts['income']:.2f}  Expense: ${amounts['expense']:.2f}  Net: ${net:.2f}")

    def list_transactions(self, limit: int = 10) -> None:
        """List recent transactions."""
        if not self.transactions:
            print("No transactions yet!")
            return

        print("\n=== Recent Transactions ===")
        for t in reversed(self.transactions[-limit:]):
            sign: str = "+" if t.type_ == "income" else "-"
            print(f"{t.date} | {t.description:20} | {sign}${t.amount:.2f} | {t.category}")

async def main() -> None:
    """Main application loop."""
    tracker = BudgetTracker()
    await tracker.load()

    while True:
        print("\n=== Budget Tracker ===")
        print(f"Current Balance: ${tracker.balance:.2f}") # Uses cached_property
        # Note: In a real app, clear the cached_property when adding new transactions
        if 'balance' in tracker.__dict__:
            del tracker.__dict__['balance']

        print("1. Add income")
        print("2. Add expense")
        print("3. View report")
        print("4. Recent transactions")
        print("5. Quit")

        choice: str = input("\nSelect: ").strip()

        if choice == "1":
            desc: str = input("Description: ")
            amount: float = float(input("Amount: "))
            category: str = input("Category: ")
            await tracker.add_transaction(desc, amount, category, "income")

        elif choice == "2":
            desc: str = input("Description: ")
            amount: float = float(input("Amount: "))
            category: str = input("Category (food/transport/entertainment/bills/other): ")
            await tracker.add_transaction(desc, amount, category, "expense")

        elif choice == "3":
            tracker.show_report()

        elif choice == "4":
            tracker.list_transactions()

        elif choice == "5":
            print("Goodbye!")
            break

        else:
            print("Invalid choice!")

if __name__ == "__main__":
    asyncio.run(main())
```

## Deep Dives

### 1. `asyncio` for File Operations
In Python, writing to disk can be a blocking operation. By using `asyncio` and custom context managers (or asynchronous file wrappers like `aiofiles`), we ensure that our UI or other background tasks (like fetching banking APIs) aren't blocked when the system persists data. The `async/await` syntax allows the event loop to schedule other tasks while waiting for I/O bounds.

### 2. Custom Context Managers
The `TimerContext` class uses `__enter__` and `__exit__` dunder methods to act as a custom context manager. This replaces bulky `try...finally` blocks for setup and teardown tasks (like closing connections, or in our case, measuring execution time for optimization).

### 3. `itertools` and `functools`
- **`functools.reduce`**: Used to accumulate totals across the transaction list without explicitly writing a `for` loop, enabling functional programming paradigms.
- **`functools.cached_property`**: Computes the balance once and caches it. In a large dataset, this saves computation. We clear the cache from `__dict__` whenever we update the transactions.
- **`itertools.groupby`**: Ideal for categorization. Instead of manual dictionary grouping in a loop, we sort the elements by our grouping key (category) and use `groupby` to efficiently yield groups.

### 4. PEP 703 (Making the Global Interpreter Lock Optional in CPython)
Python 3.13+ introduces experimental support for running Python without the GIL. In future multithreaded tracking applications where you might ingest thousands of transactions from a banking API simultaneously, No-GIL will allow true thread-parallel execution of Python bytecode. This means CPU-bound tasks, like recalculating huge transaction summaries, won't block other threads!
