## Hello, World!

The traditional first program in any language, modernized with strict type hints and docstrings:

```python
# hello.py
def main() -> None:
    """Print the traditional hello world message."""
    print("Hello, World!")

if __name__ == "__main__":
```

Save this as `hello.py` and run:

## Understanding `print()`

The `print()` function outputs text to the console.

```python
def print_examples() -> None:
    """Demonstrate different ways to use the print() function."""
    # Basic printing
    print("Hello")
    print('World')
    
    # Multiple arguments (separated by space)
    print("Hello", "World")
    
    # Custom separator
    print("Hello", "World", sep="-")  # Hello-World
    
    # Custom ending (default is newline)
    print("Hello", end=" ")
    print("World")  # Hello World
    
    # Printing variables with strict type annotations
    name: str = "Alice"
    age: int = 25
    print("Name:", name, "Age:", age)
    
    # f-strings (formatted strings) - Python 3.12+ supports nested quotes in f-strings!
    print(f"My name is {name} and I am {age} years old.")

if __name__ == "__main__":
    print_examples()
```

## Advanced Program Structure: Context Managers, Async/Await, and Collections

Modern Python relies heavily on asynchronous programming, context managers, and advanced collection types. Here is a comprehensive example incorporating Python 3.12+ features, strict typing, and standard library primitives.

### Context Managers (`@contextmanager` and `__enter__`/`__exit__`)

```python
import time
from typing import Any
from contextlib import contextmanager
from collections.abc import Generator

@contextmanager
def simple_timer() -> Generator[None, None, None]:
    """A simple generator-based context manager to measure execution time."""
    start: float = time.perf_counter()
    yield
    end: float = time.perf_counter()
    print(f"Generator Timer took {end - start:.4f} seconds.")

class ClassTimer:
    """A class-based context manager implementing __enter__ and __exit__."""
    def __init__(self) -> None:
        self.start: float = 0.0

    def __enter__(self) -> "ClassTimer":
        """Enter the runtime context and start the timer."""
        self.start = time.perf_counter()
        return self

    def __exit__(self, exc_type: Any, exc_value: Any, traceback: Any) -> bool | None:
        """Exit the runtime context and stop the timer."""
        end: float = time.perf_counter()
        print(f"Class Timer took {end - self.start:.4f} seconds.")
        return None

def run_timers() -> None:
    """Execute both types of context managers."""
    with simple_timer():
        time.sleep(0.1)
    
    with ClassTimer():
        time.sleep(0.1)

if __name__ == "__main__":
    run_timers()
```

### Async/Await Deep Dive (Python 3.12+)

With Python 3.11+, `asyncio.TaskGroup` provides modern, robust task management, replacing older `gather` patterns in many scenarios.

```python
import asyncio
from typing import Any

async def fetch_data(item_id: int) -> dict[str, int | str]:
    """
    Simulate an asynchronous data fetch.
    
    Args:
        item_id: The ID of the item to fetch.
        
    Returns:
        A dictionary containing the item data.
    """
    print(f"Fetching data for {item_id}...")
    await asyncio.sleep(0.5)
    return {"id": item_id, "status": "success"}

async def async_main() -> None:
    """Demonstrate modern asyncio with TaskGroup."""
    results: list[dict[str, int | str]] = []
    
    print("Starting TaskGroup...")
    # Python 3.11+ TaskGroup for concurrent execution
    async with asyncio.TaskGroup() as tg:
        task1 = tg.create_task(fetch_data(1))
        task2 = tg.create_task(fetch_data(2))
    
    # Tasks are awaited automatically when the TaskGroup block exits
    results.append(task1.result())
    results.append(task2.result())
    
    print(f"TaskGroup Results: {results}")
    
    # For dynamic collections, gather is still useful
    print("Starting asyncio.gather...")
    gather_results: list[dict[str, int | str]] = list(
        await asyncio.gather(*(fetch_data(i) for i in range(3, 5)))
    )
    print(f"Gather Results: {gather_results}")

if __name__ == "__main__":
    asyncio.run(async_main())
```

### Deep Dive: `itertools`, `functools`, `collections`, and Memory Profiling

Advanced functional programming primitives and memory profiling using `tracemalloc`.

```python
import itertools
import functools
import collections
import tracemalloc
from typing import Any

@functools.lru_cache(maxsize=128)
def expensive_computation(n: int) -> int:
    """An expensive recursive function heavily optimized using functools.lru_cache."""
    if n < 2:
        return n
    return expensive_computation(n - 1) + expensive_computation(n - 2)

def advanced_primitives() -> None:
    """Demonstrate itertools, functools, collections, and tracemalloc."""
    # Start tracing memory allocations
    tracemalloc.start()
    
    # 1. Collections: defaultdict and Counter
    words: list[str] = ["apple", "banana", "apple", "cherry", "banana", "apple"]
    
    counter: collections.Counter[str] = collections.Counter(words)
    print(f"Word Counts: {counter}")
    
    grouped_data: collections.defaultdict[int, list[str]] = collections.defaultdict(list)
    for word in words:
        grouped_data[len(word)].append(word)
    print(f"Grouped by Length: {dict(grouped_data)}")
    
    # 2. Itertools: combinations and chain
    letters: list[str] = ["A", "B", "C"]
    combinations: list[tuple[str, ...]] = list(itertools.combinations(letters, 2))
    print(f"Combinations: {combinations}")
    
    chained: list[str] = list(itertools.chain(["X", "Y"], ["Z"]))
    print(f"Chained: {chained}")
    
    # 3. Functools: lru_cache for memoization
    print(f"Expensive computation (cached): {expensive_computation(35)}")
    print(f"Cache info: {expensive_computation.cache_info()}")
    
    # Memory profiling results
    current: int
    peak: int
    current, peak = tracemalloc.get_traced_memory()
    print(f"Current memory usage: {current / 10**6:.4f} MB; Peak: {peak / 10**6:.4f} MB")
    
    tracemalloc.stop()

if __name__ == "__main__":
    advanced_primitives()
```

### Python GIL and Free-Threading (PEP 703)

Python 3.13 introduces experimental support for running without the Global Interpreter Lock (GIL), as detailed in PEP 703. This is a massive shift allowing true multi-threading for CPU-bound tasks in pure Python.
- **Why it matters:** Previously, CPU-bound concurrent programs in Python had to use `multiprocessing` to bypass the GIL. With free-threading, multiple threads can execute Python bytecode simultaneously.
- **How to use it:** In compatible Python builds, you can disable the GIL, leading to significant performance boosts for data-heavy and parallel workloads. Type annotations remain exactly the same, but the underlying execution model becomes fully parallel.
