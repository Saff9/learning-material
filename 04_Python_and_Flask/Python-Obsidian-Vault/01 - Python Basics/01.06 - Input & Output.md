## String Formatting

### f-strings (Recommended, Python 3.6+, Enhanced in 3.12+)

Python 3.12 allows reusing quotes inside f-strings and multiline expressions inside the braces.

```python
from typing import Any

def demonstrate_fstrings() -> None:
    """
    Shows various ways to format strings using f-strings (Python 3.12+).
    """
    name: str = "Alice"
    age: int = 25
    pi: float = 3.14159

    # Basic
    print(f"My name is {name} and I am {age} years old.")

    # Expressions inside (Python 3.12+ quote reuse)
    data: dict[str, str] = {"key": "value"}
    print(f"Look up dict value: {data['key']}")  # 3.12 allows same quotes inside

    # Formatting numbers
    print(f"PI to 2 decimal places: {pi:.2f}")      # 3.14
    print(f"PI to 4 decimal places: {pi:.4f}")      # 3.1416
    print(f"Number with padding: {42:05d}")         # 00042
    print(f"Percentage: {0.85:.1%}")                # 85.0%

    # Alignment
    print(f"{'Left':<10}|")   # Left      |
    print(f"{'Right':>10}|")  #     Right |
    print(f"{'Center':^10}|") #  Center   |

if __name__ == "__main__":
    demonstrate_fstrings()
```

## Advanced Formatting

```python
from datetime import datetime

def advanced_formatting() -> None:
    """
    Demonstrates advanced f-string capabilities including numbers and dates.
    """
    # Number formatting
    number: float = 1234567.89
    print(f"{number:,.2f}")      # 1,234,567.89
    print(f"{number:.2e}")       # 1.23e+06 (scientific)

    # Binary, octal, hex
    n: int = 42
    print(f"Binary: {n:b}")      # 101010
    print(f"Octal: {n:o}")       # 52
    print(f"Hex: {n:x}")         # 2a
    print(f"Hex: {n:X}")         # 2A

    # Date formatting
    now: datetime = datetime.now()
    print(f"Today is {now:%Y-%m-%d}")
    print(f"Time: {now:%H:%M:%S}")
    
    # Python 3.8+ debugging feature
    print(f"{n=}")               # n=42

if __name__ == "__main__":
    advanced_formatting()
```

## File I/O & Context Managers

### Basic File I/O & Memory Profiling

When dealing with large files, it is crucial to profile memory usage.

```python
import tracemalloc

def basic_file_io() -> None:
    """
    Demonstrates reading and writing files using context managers,
    while profiling memory usage.
    """
    tracemalloc.start()
    
    # Writing to a file
    with open("output.txt", "w", encoding="utf-8") as file:
        file.write("Hello, World!\n")
        file.write("This is line 2.\n")

    # Reading from a file line by line (Memory Efficient)
    with open("output.txt", "r", encoding="utf-8") as file:
        lines: list[str] = [line.strip() for line in file]
        print("Lines:", lines)
        
    current: int
    peak: int
    current, peak = tracemalloc.get_traced_memory()
    print(f"Current memory usage is {current / 1024:.2f} KB; Peak was {peak / 1024:.2f} KB")
    tracemalloc.stop()

if __name__ == "__main__":
    basic_file_io()
```

### Custom Context Managers

Context managers ensure resources are properly cleaned up. There are two ways to create them: using classes (`__enter__`/`__exit__`) and generators (`@contextmanager`).

#### Class-Based Context Manager

```python
from typing import Any, Type
from types import TracebackType

class ManagedFile:
    """
    A class-based custom context manager for handling files.
    """
    def __init__(self, filename: str, mode: str) -> None:
        self.filename: str = filename
        self.mode: str = mode
        self.file: Any | None = None

    def __enter__(self) -> Any:
        self.file = open(self.filename, self.mode, encoding="utf-8")
        return self.file

    def __exit__(
        self,
        exc_type: Type[BaseException] | None,
        exc_val: BaseException | None,
        exc_tb: TracebackType | None
    ) -> bool:
        if self.file:
            self.file.close()
        return False  # Do not suppress exceptions
```

#### Generator-Based Context Manager (using `contextlib`)

```python
from contextlib import contextmanager
from typing import Iterator, TextIO

@contextmanager
def open_managed_file(filename: str, mode: str) -> Iterator[TextIO]:
    """
    A generator-based custom context manager for handling files.
    """
    f: TextIO = open(filename, mode, encoding="utf-8")
    try:
        yield f
    finally:
        f.close()

def use_custom_context_managers() -> None:
    """
    Demonstrates both custom context managers.
    """
    with ManagedFile("test_custom1.txt", "w") as f1:
        f1.write("Using class-based context manager!\n")
        
    with open_managed_file("test_custom2.txt", "w") as f2:
        f2.write("Using generator-based context manager!\n")

if __name__ == "__main__":
    use_custom_context_managers()
```

## Deep Dive: Async I/O, TaskGroups, and GIL Evolution

For high-performance I/O, asynchronous programming with `asyncio` is essential. In modern Python, `asyncio.TaskGroup` (Python 3.11+) provides clean structured concurrency. 

**PEP 703 & The GIL:** Historically, file I/O blocks the Global Interpreter Lock (GIL). By running blocking I/O inside `asyncio.to_thread()`, we release the GIL to let the async event loop continue processing. Note that with PEP 703 (making the GIL optional in Python 3.13+), true free-threading enables multiple threads to execute Python bytecodes simultaneously, vastly improving CPU-bound performance in multi-threaded contexts while maintaining I/O efficiency.

```python
import asyncio
from typing import Any

async def fetch_data(id: int) -> dict[str, Any]:
    """
    Simulates an asynchronous data fetch operation.
    """
    print(f"Fetching data for id {id}...")
    await asyncio.sleep(1)  # Simulate non-blocking I/O wait
    return {"id": id, "data": f"Sample content for {id}"}

def blocking_io_task(filename: str, content: str) -> None:
    """
    A blocking file write operation.
    """
    with open(filename, "w", encoding="utf-8") as f:
        f.write(content)

async def process_and_save(id: int) -> None:
    """
    Fetches data asynchronously, then saves it using to_thread to prevent blocking the loop.
    """
    data: dict[str, Any] = await fetch_data(id)
    # Run blocking file I/O in a separate thread
    await asyncio.to_thread(blocking_io_task, f"async_output_{id}.txt", str(data))
    print(f"Saved data for id {id}")

async def main() -> None:
    """
    Main async entry point using Python 3.11+ TaskGroup for structured concurrency.
    """
    print("Starting TaskGroup...")
    # TaskGroup ensures all tasks finish before exiting the context block
    async with asyncio.TaskGroup() as tg:
        for i in range(1, 4):
            tg.create_task(process_and_save(i))
    
    print("All async operations completed.")

if __name__ == "__main__":
    asyncio.run(main())
```

## Data Processing with `itertools`, `functools`, and `collections`

When parsing complex output or data files, standard library primitives help construct elegant pipelines.

```python
import itertools
import functools
import collections
from typing import Iterable, Any

def process_data_pipeline() -> None:
    """
    Demonstrates processing data using advanced standard library modules.
    """
    # 1. itertools: generate an infinite counter and chain iterables
    counter: Iterable[int] = itertools.count(start=1)
    raw_data: list[str] = ["apple", "banana", "cherry"]
    extra_data: list[str] = ["date", "elderberry"]
    
    combined: list[str] = list(itertools.chain(raw_data, extra_data))
    
    # 2. collections: Count occurrences and use namedtuples
    DataRow = collections.namedtuple("DataRow", ["id", "value", "length"])
    lengths: collections.Counter[int] = collections.Counter(len(item) for item in combined)
    
    print(f"Length frequencies: {lengths}")
    
    # 3. functools: Cache expensive operations and reduce iterables
    @functools.cache
    def expensive_transform(word: str) -> str:
        return word.upper()[::-1]
    
    processed_rows: list[DataRow] = [
        DataRow(id=next(counter), value=expensive_transform(word), length=len(word))
        for word in combined
    ]
    
    print("Processed Rows:")
    for row in processed_rows:
        print(f"  {row}")

if __name__ == "__main__":
    process_data_pipeline()
```
