## Creating and Typing Lists

Modern Python relies heavily on type hints to ensure code safety and clarity.

```python
from typing import Any

# Empty list with type hint

# With elements
```

## Indexing and Slicing

```python
def process_items() -> None:
    """Demonstrate list indexing and slicing operations."""
    items: list[int] = [10, 20, 30, 40, 50]

    # Indexing
    print(items[0])    # 10
    print(items[-1])   # 50 (last)
    print(items[-2])   # 40 (second to last)

    # Slicing [start:end:step]
    print(items[1:4])    # [20, 30, 40]
    print(items[:3])     # [10, 20, 30]
    print(items[2:])     # [30, 40, 50]
    print(items[::2])    # [10, 30, 50]
    print(items[::-1])   # [50, 40, 30, 20, 10] (reversed)

if __name__ == "__main__":
    process_items()
```

## Modifying Lists and Advanced Collections

```python
from collections import deque

def modify_collections() -> None:
    """Demonstrate list modifications and deque usage."""
    fruits: list[str] = ["apple"]
    
    # Append and Extend
    fruits.append("banana")
    fruits.extend(["cherry", "date"])
    
    # Insert, Remove, Pop
    fruits.insert(1, "apricot")
    fruits.remove("banana")
    last: str = fruits.pop()
    
    # Using deque for O(1) left insertions/pops
    queue: deque[str] = deque(fruits)
    queue.appendleft("mango")
    print(list(queue))

if __name__ == "__main__":
    modify_collections()
```

## Deep Dive: Asynchronous List Processing (`asyncio`)

```python
import asyncio

async def fetch_data(item: int) -> str:
    """Simulate an async I/O operation."""
    await asyncio.sleep(0.1)
    return f"Processed {item}"

async def process_list_async(items: list[int]) -> list[str]:
    """Asynchronously process a list of items using gather."""
    tasks = [fetch_data(item) for item in items]
    results: list[str] = await asyncio.gather(*tasks)
    return results

if __name__ == "__main__":
    # Python 3.12+ standard async entry point
    results: list[str] = asyncio.run(process_list_async([1, 2, 3, 4, 5]))
    print(results)
```

## Custom Context Managers for Lists

```python
from contextlib import contextmanager
from typing import Iterator, Any

@contextmanager
def temporary_append(lst: list[Any], item: Any) -> Iterator[list[Any]]:
    """Context manager to temporarily append an item to a list."""
    lst.append(item)
    try:
        yield lst
    finally:
        lst.pop()

def test_context_manager() -> None:
    """Test the temporary_append context manager."""
    my_list: list[int] = [1, 2]
    with temporary_append(my_list, 3) as temp_list:
        print(f"Inside context: {temp_list}")
    print(f"Outside context: {my_list}")

if __name__ == "__main__":
    test_context_manager()
```

## Deep Dive: Memory Profiling and Python GIL (PEP 703)

### Memory Profiling (`tracemalloc`)

Python lists are dynamic arrays, which means they over-allocate memory to allow for efficient appends.

```python
import tracemalloc

def profile_list_allocation() -> None:
    """Profile memory usage during list creation."""
    tracemalloc.start()
    
    large_list: list[int] = [i for i in range(1_000_000)]
    
    current, peak = tracemalloc.get_traced_memory()
    print(f"Current memory usage: {current / 10**6:.2f} MB")
    print(f"Peak memory usage: {peak / 10**6:.2f} MB")
    
    tracemalloc.stop()

if __name__ == "__main__":
    profile_list_allocation()
```

### PEP 703 and the GIL
Historically, the Global Interpreter Lock (GIL) prevents multiple native threads from executing Python bytecodes simultaneously. This meant that pure CPU-bound list operations wouldn't speed up with simple threading. 
With PEP 703 (Making the Global Interpreter Lock Optional in CPython) introduced in the Python 3.13 timeframe (and shaping 3.12 discussions), free-threading builds allow multithreaded list processing without GIL contention. Thread-safe operations on lists (or using concurrent lock-free structures) are becoming increasingly important.

### Exercise 3: Flatten List
```python
def flatten_list(nested: list[list[int]]) -> list[int]:
    """Flatten a 2D list into a 1D list."""
    return [x for sublist in nested for x in sublist]

if __name__ == "__main__":
    print(flatten_list([[1, 2], [3, 4], [5, 6]]))
```

### Exercise 4: Find Second Largest
```python
def second_largest(numbers: list[int]) -> int | None:
    """Find the second largest number in a list."""
    unique: list[int] = list(set(numbers))
    unique.sort()
    return unique[-2] if len(unique) >= 2 else None

if __name__ == "__main__":
    print(second_largest([10, 20, 4, 45, 99]))
```
