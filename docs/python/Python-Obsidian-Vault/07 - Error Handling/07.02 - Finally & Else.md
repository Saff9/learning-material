## The `else` Clause

```python
def process_number() -> None:
    """Demonstrates the try-except-else pattern."""
    try:
        number_str: str = input("Enter a number: ")
        number: int = int(number_str)
    except ValueError:
        print("That is not a number!")
    else:
        # Runs ONLY if no exception occurred
        print(f"You entered: {number}")
        result: int = number * 2
        print(f"Double: {result}")

# process_number()
```

## The `finally` Clause

```python
def read_config(filename: str) -> str | None:
    """Demonstrates finally for resource cleanup without context managers."""
    file = None
    try:
        file = open(filename, "r")
        content: str = file.read()
        return content
    except FileNotFoundError:
        print("File not found!")
        return None
    finally:
        # ALWAYS runs, even if an exception occurred or function returned
        print("Cleaning up resources...")
        if file:
            file.close()

# read_config("missing.txt")
```

## Complete Structure

```python
def complex_operation() -> None:
    """Shows the full try-except-else-finally flow."""
    try:
        print("Trying...")
        result: float = 10 / 2
    except ZeroDivisionError:
        print("Division by zero!")
    else:
        print(f"Success! Result: {result}")
    finally:
        print("This always runs")

complex_operation()
```

## Context Managers: A Better Alternative to Finally
While `finally` is useful, Python's `with` statement and context managers provide a more elegant and reusable way to guarantee cleanup.

### 1. Class-Based Context Manager

```python
from typing import Any

class DatabaseConnection:
    """A custom context manager for a simulated database connection."""
    def __init__(self, db_url: str) -> None:
        self.db_url = db_url

    def __enter__(self) -> 'DatabaseConnection':
        """Setup code, executed at the start of the with block."""
        print(f"Connecting to database at {self.db_url}")
        return self

    def __exit__(self, exc_type: type | None, exc_val: Exception | None, exc_tb: Any) -> bool:
        """Cleanup code, executed at the end of the with block."""
        print(f"Closing database connection to {self.db_url}")
        # Return False to propagate exceptions, True to suppress them
        if exc_type:
            print(f"An error occurred: {exc_val}")
        return False

def use_db() -> None:
    with DatabaseConnection("postgresql://localhost:5432") as db:
        print("Executing query...")
        # If an error happens here, __exit__ still runs

use_db()
```

### 2. Generator-Based Context Manager
Using `contextlib.contextmanager` is often shorter for simple resources.

```python
from contextlib import contextmanager
from collections.abc import Generator

@contextmanager
def open_file_safe(path: str, mode: str) -> Generator[Any, None, None]:
    """A custom context manager for file handling using generators."""
    file = None
    try:
        file = open(path, mode)
        yield file
    finally:
        if file:
            print(f"Closing file: {path}")
            file.close()

def write_data() -> None:
    try:
        with open_file_safe("temp.txt", "w") as f:
            f.write("Hello, World!")
    except IOError as e:
        print(f"IO Error: {e}")

write_data()
```

### 3. Asynchronous Context Managers
For async programming (e.g., async database drivers, HTTP sessions), you need `__aenter__` and `__aexit__`.

```python
import asyncio

class AsyncSession:
    """An asynchronous context manager."""
    async def __aenter__(self) -> 'AsyncSession':
        print("Acquiring async lock/session...")
        await asyncio.sleep(0.1)
        return self

    async def __aexit__(self, exc_type: type | None, exc_val: Exception | None, exc_tb: Any) -> bool:
        print("Releasing async lock/session...")
        await asyncio.sleep(0.1)
        return False

async def fetch_async() -> None:
    async with AsyncSession() as session:
        print("Fetching data asynchronously...")

asyncio.run(fetch_async())
```

### Exercise 1: Calculator with Cleanup
```python
def calculator() -> None:
    """A safe calculator utilizing try-except-else-finally."""
    while True:
        try:
            user_input: str = input("First number (or 'q' to quit): ")
            if user_input.lower() == 'q':
                break
            a: float = float(user_input)
            b: float = float(input("Second number: "))
            op: str = input("Operation (+, -, *, /): ")
        except ValueError:
            print("Invalid input. Try again.")
            continue

        try:
            match op:
                case "+": result = a + b
                case "-": result = a - b
                case "*": result = a * b
                case "/": result = a / b
                case _:
                    print("Unknown operation")
                    continue
        except ZeroDivisionError:
            print("Cannot divide by zero!")
        else:
            print(f"Result: {result}")
        finally:
            print("---")

# calculator()
```
