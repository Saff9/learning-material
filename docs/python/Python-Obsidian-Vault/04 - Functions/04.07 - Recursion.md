## Basic Recursion

```python
def countdown(n: int) -> None:
    """Recursively counts down to zero."""
    if n <= 0:  # Base case
        print("Blast off!")
    else:       # Recursive case
        print(n)
        countdown(n - 1)

if __name__ == "__main__":
    countdown(5)
```

## Memoization with `functools.cache`

Naive recursion can be extremely slow for branching problems like Fibonacci due to recalculating identical states. `functools.cache` provides automatic memoization.

```python
from functools import cache
import time

@cache
def fibonacci(n: int) -> int:
    """Returns the nth Fibonacci number in O(n) time using memoization."""
    if n <= 1:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

if __name__ == "__main__":
    start = time.perf_counter()
    print(f"Fibonacci(100): {fibonacci(100)}")
    print(f"Time taken (cached): {time.perf_counter() - start:.6f}s")
```

## Recursive Search

```python
from typing import Sequence

def binary_search(arr: Sequence[int | float], target: int | float, low: int = 0, high: int | None = None) -> int:
    """Performs recursive binary search on a sorted sequence."""
    if high is None:
        high = len(arr) - 1

    if low > high:
        return -1

    mid = (low + high) // 2

    if arr[mid] == target:
        return mid
    elif arr[mid] > target:
        return binary_search(arr, target, low, mid - 1)
    else:
        return binary_search(arr, target, mid + 1, high)

if __name__ == "__main__":
    print(binary_search([1, 3, 5, 7, 9], 5))  # Returns 2
```

## Recursion Limit and Profiling

Python lacks standard tail-call optimization (TCO). This means deep recursions will trigger a `RecursionError`. You can inspect and modify this limit with the `sys` module, though it should be done with caution.

```python
import sys

def check_recursion_limit() -> None:
    """Inspects the current Python recursion depth limit."""
    limit = sys.getrecursionlimit()
    print(f"Current recursion limit: {limit}")
    
    # You can increase it if strictly necessary for algorithms:
    # sys.setrecursionlimit(2000)

if __name__ == "__main__":
    check_recursion_limit()
```

## 🧪 Practice

### Exercise 1: Sum of List
```python
from typing import Sequence

def recursive_sum(numbers: Sequence[int | float]) -> int | float:
    """Recursively sums the elements of a sequence."""
    if not numbers:
        return 0
    return numbers[0] + recursive_sum(numbers[1:])

if __name__ == "__main__":
    print(recursive_sum([1, 2, 3, 4, 5]))  # 15
```

### Exercise 2: String Reversal
```python
def reverse_string(s: str) -> str:
    """Recursively reverses a string."""
    if len(s) <= 1:
        return s
    return reverse_string(s[1:]) + s[0]

if __name__ == "__main__":
    print(reverse_string("hello"))  # "olleh"
```

### Exercise 3: Power Function
```python
def power(base: int | float, exp: int) -> int | float:
    """Recursively computes base raised to the exp power."""
    if exp == 0:
        return 1
    return base * power(base, exp - 1)

if __name__ == "__main__":
    print(power(2, 10))  # 1024
```

### Exercise 4: Palindrome Check
```python
def is_palindrome(s: str) -> bool:
    """Recursively checks if a string is a palindrome."""
    if len(s) <= 1:
        return True
    if s[0] != s[-1]:
        return False
    return is_palindrome(s[1:-1])

if __name__ == "__main__":
    print(is_palindrome("racecar"))  # True
```
