```python
# password_generator.py
"""A secure and modern password generator in Python 3.12+."""

import asyncio
import itertools
import functools
import secrets
import string
from typing import Tuple, List, Generator

def validate_criteria(func):
    """Decorator to validate that at least one character set is selected."""
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        use_lower = kwargs.get('use_lower', True)
        use_upper = kwargs.get('use_upper', True)
        use_digits = kwargs.get('use_digits', True)
        use_special = kwargs.get('use_special', True)
        
        if not any([use_lower, use_upper, use_digits, use_special]):
            raise ValueError("At least one character type must be selected")
        return func(*args, **kwargs)
    return wrapper

class PasswordContext:
    """A context manager to temporarily set strict password policies."""
    def __init__(self, length: int = 12):
        self.length = length

    def __enter__(self) -> 'PasswordContext':
        print(f"Entering secure password generation context (length={self.length})...")
        return self

    def __exit__(self, exc_type, exc_value, traceback) -> None:
        print("Exiting secure context...")

@validate_criteria
def generate_password(
    length: int = 12, 
    use_upper: bool = True, 
    use_lower: bool = True, 
    use_digits: bool = True, 
    use_special: bool = True
) -> str:
    """
    Generate a cryptographically secure random password.

    Args:
        length: Length of the password.
        use_upper: Include uppercase letters.
        use_lower: Include lowercase letters.
        use_digits: Include digits.
        use_special: Include special characters.

    Returns:
        The generated password string.
    """
    char_sets: List[str] = []

    if use_lower:
        char_sets.append(string.ascii_lowercase)
    if use_upper:
        char_sets.append(string.ascii_uppercase)
    if use_digits:
        char_sets.append(string.digits)
    if use_special:
        char_sets.append(string.punctuation)

    # Flatten the character sets using itertools
    characters: str = ''.join(itertools.chain(char_sets))

    # Use secrets for cryptographically secure random
    password: str = ''.join(secrets.choice(characters) for _ in range(length))
    return password

def check_strength(password: str) -> Tuple[str, List[str]]:
    """
    Check password strength against multiple criteria.

    Args:
        password: The password to evaluate.

    Returns:
        A tuple containing the strength level and a list of feedback messages.
    """
    score: int = 0
    feedback: List[str] = []

    if len(password) >= 8:
        score += 1
    else:
        feedback.append("Password should be at least 8 characters")

    if any(c.isupper() for c in password):
        score += 1
    else:
        feedback.append("Add uppercase letters")

    if any(c.islower() for c in password):
        score += 1
    else:
        feedback.append("Add lowercase letters")

    if any(c.isdigit() for c in password):
        score += 1
    else:
        feedback.append("Add digits")

    if any(c in string.punctuation for c in password):
        score += 1
    else:
        feedback.append("Add special characters")

    strength_levels: List[str] = ["Very Weak", "Weak", "Fair", "Good", "Strong", "Very Strong"]
    return strength_levels[score], feedback

async def generate_passphrase_async(num_words: int = 4, separator: str = "-") -> str:
    """
    Generate a memorable passphrase asynchronously.
    Demonstrates async/await pattern for potential I/O bound tasks like fetching wordlists.
    """
    word_list: List[str] = [
        "apple", "banana", "cherry", "delta", "eagle", "forest",
        "garden", "house", "island", "jungle", "king", "lemon",
        "mountain", "night", "ocean", "piano", "queen", "river",
        "sun", "tiger", "umbrella", "violet", "water", "yellow"
    ]
    
    # Simulate an I/O delay (e.g., fetching words from an external secure API)
    await asyncio.sleep(0.1)

    words: List[str] = [secrets.choice(word_list) for _ in range(num_words)]
    return separator.join(words)

async def main() -> None:
    print("=== Modern Password Generator ===")
    
    with PasswordContext(length=16):
        pwd: str = generate_password(length=16)
        print(f"\nPassword: {pwd}")
        strength, feedback = check_strength(pwd)
        print(f"Strength: {strength}")

    print("\nGenerating passphrase asynchronously...")
    passphrase: str = await generate_passphrase_async()
    print(f"Passphrase: {passphrase}")

if __name__ == "__main__":
    asyncio.run(main())
```

## Deep Dives

### 1. `async`/`await` Patterns
The `generate_passphrase_async` function uses the `async def` syntax. This allows for non-blocking operations. In a real-world scenario, you might fetch a massive dictionary of words from an external API or read a very large file from disk asynchronously, allowing other operations to continue without blocking the main execution thread.

### 2. Custom Context Managers
The `PasswordContext` class acts as a context manager (using `__enter__` and `__exit__`). Context managers are ideal for setting up temporary states, acquiring resources, and ensuring they are cleanly released or reset once the block of code inside the `with` statement finishes. Here, it signifies the generation in a secure block.

### 3. `itertools` and `functools`
- **`functools.wraps`**: We used `@functools.wraps(func)` inside the `validate_criteria` decorator. This preserves the metadata of the original function (like its docstring and name) so that the wrapped function doesn't lose its identity, which is crucial for modern typing and IDE support.
- **`itertools.chain`**: In `generate_password`, `itertools.chain` is used to elegantly flatten an array of strings into a single iterable without manual loops, demonstrating a clean functional approach to processing collections.

### 4. PEP 703 (Making the Global Interpreter Lock Optional in CPython)
For applications that generate thousands of passwords or passphrases simultaneously for large user bases, the Global Interpreter Lock (GIL) in CPython has traditionally been a bottleneck, as it prevents multiple native threads from executing Python bytecodes concurrently.
PEP 703 proposes making the GIL optional. Once fully integrated into CPython (starting from Python 3.13+ experimentally), CPU-bound tasks like generating large numbers of cryptographically secure strings could take full advantage of multi-core processors without resorting to multiprocessing, fundamentally changing how high-performance parallel Python code is written.
