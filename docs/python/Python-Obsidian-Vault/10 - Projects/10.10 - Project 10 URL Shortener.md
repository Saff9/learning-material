## Full Source Code

This modernized implementation of a URL shortener brings it up to Python 3.12+ standards. We utilize `asyncio` for non-blocking file operations, custom context managers for simulated file locking, and strict type hints throughout.

```python
# url_shortener.py
"""
A modern, asynchronous URL shortener with file-based storage, 
featuring strict type annotations and custom context managers.
"""

import asyncio
import hashlib
import json
import re
import itertools
from datetime import datetime
from pathlib import Path
from typing import TypedDict, Optional, Any
from functools import lru_cache

# Python 3.12+ TypedDict for strict structural typing
class URLData(TypedDict):
    """Type definition for URL metadata."""
    url: str
    created: str
    clicks: int

class FileLockContextManager:
    """
    A custom asynchronous context manager to simulate file locking.
    Ensures safe concurrent access to the underlying JSON store.
    """
    def __init__(self, filename: Path) -> None:
        self.lock_file = filename.with_suffix('.lock')

    async def __aenter__(self) -> 'FileLockContextManager':
        # Simulate acquiring a lock asynchronously
        while self.lock_file.exists():
            await asyncio.sleep(0.01)
        self.lock_file.touch()
        return self

    async def __aexit__(self, exc_type: Any, exc_val: Any, exc_tb: Any) -> None:
        # Simulate releasing the lock
        if self.lock_file.exists():
            self.lock_file.unlink()

class URLShortener:
    """Async URL Shortener Service."""
    
    def __init__(self, filename: str | Path = "urls.json") -> None:
        self.filename = Path(filename)
        self.urls: dict[str, URLData] = {}

    async def load(self) -> None:
        """Asynchronously load URL mappings from file."""
        async with FileLockContextManager(self.filename):
            if self.filename.exists():
                # Utilizing asyncio.to_thread to prevent blocking the event loop on I/O
                content = await asyncio.to_thread(self.filename.read_text, encoding='utf-8')
                self.urls = json.loads(content) if content else {}

    async def save(self) -> None:
        """Asynchronously save URL mappings to file."""
        async with FileLockContextManager(self.filename):
            data = json.dumps(self.urls, indent=2)
            await asyncio.to_thread(self.filename.write_text, data, encoding='utf-8')

    @staticmethod
    @lru_cache(maxsize=128)
    def validate_url(url: str) -> bool:
        """
        Check if URL is valid.
        Uses functools.lru_cache to memoize results of frequent URL validations,
        improving performance on repetitive lookups.
        """
        pattern = re.compile(r"^https?://[^\s/$.?#].[^\s]*$")
        return pattern.match(url) is not None

    def generate_code(self, url: str) -> str:
        """Generate a deterministic 6-character short code for a URL."""
        return hashlib.md5(url.encode()).hexdigest()[:6]

    async def shorten(self, url: str) -> Optional[str]:
        """Shorten a URL asynchronously and save to store."""
        if not self.validate_url(url):
            print("Invalid URL! Must start with http:// or https://")
            return None

        # Check if already shortened
        for code, data in self.urls.items():
            if data["url"] == url:
                print(f"URL already shortened: {code}")
                return code

        code = self.generate_code(url)

        # Handle collisions efficiently using itertools.count
        if code in self.urls:
            counter = itertools.count(1)
            while code in self.urls:
                next_val = next(counter)
                code = hashlib.md5(f"{url}{next_val}".encode()).hexdigest()[:6]

        self.urls[code] = {
            "url": url,
            "created": datetime.now().isoformat(),
            "clicks": 0
        }
        
        await self.save()
        print(f"Shortened: {url} -> {code}")
        return code

    async def expand(self, code: str) -> Optional[str]:
        """Get original URL from short code, asynchronously incrementing click tracking."""
        if code in self.urls:
            self.urls[code]["clicks"] += 1
            await self.save()
            return self.urls[code]["url"]
        return None

    def get_stats(self, code: str) -> Optional[URLData]:
        """Get statistics for a short URL using Python 3.8+ Walrus Operator."""
        # Using walrus operator (:=) for concise assignment and checking
        if data := self.urls.get(code):
            print(f"\nURL: {data['url']}")
            print(f"Created: {data['created']}")
            print(f"Clicks: {data['clicks']}")
            return data
        
        print("Code not found!")
        return None

async def main() -> None:
    """Main interactive asynchronous execution."""
    shortener = URLShortener()
    await shortener.load()

    # Simple runnable example
    print("Testing Shortener...")
    code = await shortener.shorten("https://www.python.org/downloads/release/python-3120/")
    if code:
        expanded = await shortener.expand(code)
        print(f"Expanded: {expanded}")
        shortener.get_stats(code)

if __name__ == "__main__":
    # Standard entry point for modern async Python scripts
    asyncio.run(main())
```

## Deep Dives

### Async/Await & Custom Context Managers
In web applications (like a Flask or FastAPI frontend that might wrap this), file I/O can severely block the main thread. 
- By wrapping file reads/writes in `asyncio.to_thread`, we keep the event loop free to handle other requests. 
- The `FileLockContextManager` utilizes the `__aenter__` and `__aexit__` dunder methods to create an *asynchronous context manager*. This prevents race conditions where two simultaneous async tasks try to write to the JSON file at the same time, potentially corrupting it.

### `functools.lru_cache` and `itertools`
- **`functools.lru_cache`**: Regular expressions can be computationally expensive. We decorated `validate_url` with `@lru_cache(maxsize=128)`. If the same URL is validated multiple times, Python instantly returns the cached boolean without recompiling or executing the regex.
- **`itertools.count`**: During hash collision handling, we needed a way to deterministically mutate the input until we find a unique hash. `itertools.count(1)` acts as a fast, infinite generator, yielding consecutive integers that we append to the URL string.

### PEP 703 and the Optional GIL
PEP 703 introduces the capability to run CPython *without* the Global Interpreter Lock (GIL) (often referred to as "Free-threading" in Python 3.13+). 
- **Relevance**: In a traditional URL shortener, concurrent user requests are constrained to a single thread executing Python bytecode at any given moment due to the GIL. 
- With PEP 703 enabled, multi-threaded frameworks wrapping this URL shortener will genuinely achieve parallel execution across multiple CPU cores. The `FileLockContextManager` implemented above becomes even more critical in a free-threaded environment, as hardware-level race conditions on shared file states will manifest more frequently than in GIL-bound environments.
