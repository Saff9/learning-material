## Full Source Code

The following example is a modernized implementation. It introduces an asynchronous custom context manager (`TodoContextManager`) to handle loading and saving tasks automatically, ensuring file operations do not block the main event loop.

```python
# todo.py
"""
A modernized command-line todo list application utilizing Python 3.12+ features.
Demonstrates async/await, dataclasses, context managers, and strict type annotations.
"""

import asyncio
import json
import logging
from dataclasses import dataclass, asdict
from datetime import datetime
from pathlib import Path
from typing import Any
import itertools
import functools

# Configure basic logging
logging.basicConfig(level=logging.INFO, format="%(levelname)s: %(message)s")

@dataclass
class Task:
    """Represents a single Todo task."""
    id: int
    title: str
    priority: str = "medium"
    completed: bool = False
    created: str = ""

    def __post_init__(self) -> None:
        """Initialize creation timestamp if not provided."""
        if not self.created:
            self.created = datetime.now().isoformat()

class TodoContextManager:
    """
    A custom asynchronous context manager for safely managing file state.
    Ensures that tasks are properly loaded before execution and saved afterward.
    """
    def __init__(self, filename: Path | str) -> None:
        self.filename: Path = Path(filename)
        self.tasks: list[Task] = []

    async def __aenter__(self) -> 'TodoContextManager':
        """Load tasks from disk asynchronously."""
        if self.filename.exists():
            # Run file IO in a separate thread to avoid blocking the event loop
            data = await asyncio.to_thread(self.filename.read_text, encoding="utf-8")
            if data.strip():
                raw_tasks = json.loads(data)
                self.tasks = [Task(**task) for task in raw_tasks]
        return self

    async def __aexit__(self, exc_type: Any, exc_val: Any, exc_tb: Any) -> None:
        """Save tasks back to disk upon exiting the context block."""
        data = json.dumps([asdict(t) for t in self.tasks], indent=2)
        await asyncio.to_thread(self.filename.write_text, data, encoding="utf-8")
        if exc_type:
            logging.error(f"Error during context management: {exc_val}")


class TodoList:
    """Manages CRUD operations for Todo Tasks asynchronously."""
    
    def __init__(self, filename: str = "todos.json") -> None:
        self.filename: Path = Path(filename)
    
    async def add(self, title: str, priority: str = "medium") -> None:
        """Add a new task."""
        async with TodoContextManager(self.filename) as ctx:
            # Generate a new ID dynamically based on existing maximum
            next_id = max((t.id for t in ctx.tasks), default=0) + 1
            task = Task(id=next_id, title=title, priority=priority)
            ctx.tasks.append(task)
            print(f"Added: {title}")

    async def list_tasks(self, show_all: bool = False) -> None:
        """List all tasks, applying functional programming filters."""
        async with TodoContextManager(self.filename) as ctx:
            if not ctx.tasks:
                print("No tasks found!")
                return
            
            print("\n=== Your Tasks ===")
            
            # Using functional programming patterns (filter and functools.partial)
            def is_visible(task: Task, all_tasks: bool) -> bool:
                return all_tasks or not task.completed
                
            visible_filter = functools.partial(is_visible, all_tasks=show_all)
            
            for task in filter(visible_filter, ctx.tasks):
                status = "[X]" if task.completed else "[ ]"
                print(f"{status} {task.id}. {task.title} ({task.priority})")

    async def complete(self, task_id: int) -> None:
        """Mark a task as completed."""
        async with TodoContextManager(self.filename) as ctx:
            for task in ctx.tasks:
                if task.id == task_id:
                    task.completed = True
                    print(f"Completed: {task.title}")
                    return
            print("Task not found!")

    async def delete(self, task_id: int) -> None:
        """Delete a task using itertools."""
        async with TodoContextManager(self.filename) as ctx:
            # Using itertools.filterfalse to remove the task efficiently
            original_len = len(ctx.tasks)
            ctx.tasks = list(itertools.filterfalse(lambda t: t.id == task_id, ctx.tasks))
            
            if len(ctx.tasks) < original_len:
                print(f"Deleted task ID: {task_id}")
            else:
                print("Task not found!")

async def main() -> None:
    """Main application loop."""
    todo = TodoList()

    while True:
        print("\n=== Todo List ===")
        print("1. Add task")
        print("2. List tasks")
        print("3. Complete task")
        print("4. Delete task")
        print("5. Show all (including completed)")
        print("6. Quit")

        choice: str = input("\nSelect: ").strip()

        if choice == "1":
            title: str = input("Task title: ")
            priority: str = input("Priority (low/medium/high): ") or "medium"
            await todo.add(title, priority)
        elif choice == "2":
            await todo.list_tasks()
        elif choice == "3":
            try:
                task_id = int(input("Task ID to complete: "))
                await todo.complete(task_id)
            except ValueError:
                print("Invalid ID!")
        elif choice == "4":
            try:
                task_id = int(input("Task ID to delete: "))
                await todo.delete(task_id)
            except ValueError:
                print("Invalid ID!")
        elif choice == "5":
            await todo.list_tasks(show_all=True)
        elif choice == "6":
            print("Goodbye!")
            break
        else:
            print("Invalid choice!")

if __name__ == "__main__":
    asyncio.run(main())
```

## Deep Dives

### 1. Asynchronous I/O and `async` / `await`
File operations (`read_text`, `write_text`) are typically blocking. By using `asyncio.to_thread()`, we execute them in a separate thread, freeing the main thread to run the `asyncio` event loop. This non-blocking IO design is critical for scalability in web applications or UI-driven systems.

### 2. Custom Context Managers
The `TodoContextManager` defines `__aenter__` and `__aexit__`. This ensures that every time we manipulate tasks (Add, Complete, Delete), we safely acquire the current list from the disk and reliably save back any modifications, even if an exception occurs during execution. It abstracts state management beautifully.

### 3. Functional Iterables (`itertools` & `functools`)
- **`functools.partial`**: Used in `list_tasks` to create a pre-configured version of the `is_visible` function. This makes filtering dynamic, reusable, and cleaner.
- **`itertools.filterfalse`**: Used in `delete` to elegantly filter out the task to be deleted without writing verbose `for` loops or list comprehensions, promoting functional iteration.

### 4. PEP 703 and The No-GIL Future
In Python 3.13+, **PEP 703** introduces a build of Python without the Global Interpreter Lock (GIL). 
While our script uses `asyncio.to_thread` for I/O operations (which already releases the GIL naturally in CPython), PEP 703 will allow CPU-bound extensions of this script (like searching/sorting millions of tasks simultaneously) to run truly in parallel on multiple cores without requiring the heavy overhead of the `multiprocessing` module.

## Common Bugs and Fixes

| Bug | Cause | Fix |
|-----|-------|-----|
| Tasks not saved | Exceptions blocking `__aexit__` | Handle exceptions gracefully inside `__aexit__` or the async context |
| `asyncio.run()` RuntimeError | Loop already running | Avoid nested `asyncio.run()` calls in modern environments like Jupyter |
| Missing File Error | Initializing without file creation | `Path.exists()` check ensures safe reading |
