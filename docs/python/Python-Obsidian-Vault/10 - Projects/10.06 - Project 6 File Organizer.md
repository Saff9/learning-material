```python
# file_organizer.py
"""
Organizes files in a directory asynchronously by their extensions or dates.
Requires Python 3.12+.
"""

import asyncio
import os
import shutil
import time
from collections import defaultdict
from collections.abc import AsyncGenerator, Callable
from functools import partial
from itertools import groupby
from pathlib import Path
from typing import Any


class TimerContext:
    """Custom context manager to time execution duration."""
    
    def __init__(self, operation_name: str) -> None:
        self.operation_name: str = operation_name
        self.start_time: float = 0.0

    def __enter__(self) -> 'TimerContext':
        self.start_time = time.perf_counter()
        print(f"[{self.operation_name}] Started...")
        return self

    def __exit__(self, exc_type: Any, exc_val: Any, exc_tb: Any) -> None:
        elapsed = time.perf_counter() - self.start_time
        if exc_type:
            print(f"[{self.operation_name}] Failed with {exc_type.__name__}")
        else:
            print(f"[{self.operation_name}] Completed in {elapsed:.4f} seconds.")


class FileOrganizer:
    """Class to manage directory organization."""

    def __init__(self, source_dir: str | Path) -> None:
        self.source_dir: Path = Path(source_dir).resolve()
        self.stats: defaultdict[str, int] = defaultdict(int)

    async def _move_file(self, file_path: Path, target_dir: Path, category: str) -> None:
        """
        Asynchronously move a single file using a thread pool.
        This prevents blocking the event loop during I/O operations.
        """
        target_dir.mkdir(parents=True, exist_ok=True)
        target_path: Path = target_dir / file_path.name

        # Handle duplicates
        counter: int = 1
        while target_path.exists():
            stem: str = file_path.stem
            suffix: str = file_path.suffix
            target_path = target_dir / f"{stem}_{counter}{suffix}"
            counter += 1

        # Use asyncio.to_thread for I/O bound blocking calls
        await asyncio.to_thread(shutil.move, str(file_path), str(target_path))
        self.stats[category] += 1
        print(f"Moved: {file_path.name} -> {category}/")

    async def organize_by_extension(self) -> None:
        """Organize files into folders by their file extension asynchronously."""
        if not self.source_dir.exists():
            print(f"Directory {self.source_dir} does not exist!")
            return

        with TimerContext("Organize by Extension"):
            tasks: list[asyncio.Task[None]] = []
            
            for file_path in self.source_dir.iterdir():
                if file_path.is_file():
                    ext: str = file_path.suffix.lower()
                    category: str = ext[1:] if ext else "no_extension"
                    target_dir: Path = self.source_dir / category
                    
                    tasks.append(
                        asyncio.create_task(self._move_file(file_path, target_dir, category))
                    )
            
            if tasks:
                await asyncio.gather(*tasks)

        self.print_stats()

    async def organize_by_date(self) -> None:
        """Organize files by their modification date (YYYY-MM)."""
        if not self.source_dir.exists():
            return

        with TimerContext("Organize by Date"):
            tasks: list[asyncio.Task[None]] = []
            
            # Using itertools.groupby as a demonstration
            # Group files by their YYYY-MM modification date
            files: list[Path] = [p for p in self.source_dir.iterdir() if p.is_file()]
            
            def get_mtime_str(p: Path) -> str:
                mtime = p.stat().st_mtime
                return time.strftime("%Y-%m", time.localtime(mtime))
                
            # groupby requires sorted input
            files.sort(key=get_mtime_str)
            
            for date_folder, grouped_files in groupby(files, key=get_mtime_str):
                target_dir: Path = self.source_dir / date_folder
                for file_path in grouped_files:
                    tasks.append(
                        asyncio.create_task(self._move_file(file_path, target_dir, date_folder))
                    )
            
            if tasks:
                await asyncio.gather(*tasks)
                
        self.print_stats()

    def clean_empty_folders(self) -> None:
        """Synchronous method to remove empty folders recursively."""
        with TimerContext("Clean Empty Folders"):
            removed: int = 0
            for dir_path in list(self.source_dir.rglob("*")):
                if dir_path.is_dir() and not any(dir_path.iterdir()):
                    dir_path.rmdir()
                    removed += 1
            print(f"Removed {removed} empty folders")

    def print_stats(self) -> None:
        """Print organization statistics using functools."""
        print("\n=== Organization Complete ===")
        total: int = sum(self.stats.values())
        print(f"Total files organized: {total}")
        
        # Using a partial function for formatted printing
        print_stat: Callable[[str, int], None] = partial(
            lambda k, v: print(f"  [{k}]: {v} files")
        )
        
        for ext, count in sorted(self.stats.items()):
            print_stat(ext, count)


async def main() -> None:
    """Main async entry point."""
    print("=== File Organizer ===")
    directory: str = input("Enter directory to organize (or press enter for current): ").strip()
    
    if not directory:
        directory = "."

    organizer: FileOrganizer = FileOrganizer(directory)

    print("\n1. Organize by extension")
    print("2. Organize by date")
    print("3. Clean empty folders")

    choice: str = input("\nSelect: ").strip()

    if choice == "1":
        await organizer.organize_by_extension()
    elif choice == "2":
        await organizer.organize_by_date()
    elif choice == "3":
        organizer.clean_empty_folders()
    else:
        print("Invalid choice!")


if __name__ == "__main__":
    asyncio.run(main())
```

## Deep Dives

### 1. Context Managers (Custom `TimerContext`)
Context managers in Python (used with the `with` statement) elegantly handle resource allocation and release. Our `TimerContext` implements `__enter__` and `__exit__` dunder methods to calculate block execution time seamlessly, ensuring the timer finishes even if an exception occurs inside the block.

### 2. Async/Await and `asyncio.to_thread`
File I/O in standard Python (`shutil.move`) is blocking. Running this synchronously pauses the entire program until the disk finishes writing. 
By delegating to `asyncio.to_thread()`, we run the blocking `shutil.move` in a separate thread. `asyncio.gather(*tasks)` then allows us to initiate hundreds of file moves simultaneously, drastically improving performance for directories with many files.

### 3. Functional Magic: `itertools` and `functools`
- **`itertools.groupby`**: Groups elements based on a key function. In our code, we use it to chunk files by their modification date (YYYY-MM). *Note: `groupby` requires the input iterable to be sorted by the same key first.*
- **`functools.partial`**: Allows you to "freeze" some arguments of a function or create specialized wrapper functions. We used it to create a custom `print_stat` function dedicated to standardizing output format.

### 4. PEP 703: Making the Global Interpreter Lock (GIL) Optional
Historically, Python's GIL prevents multiple native threads from executing Python bytecodes concurrently.
- **The Problem:** In CPU-bound multithreaded Python code, threads bottleneck on the GIL. 
- **The PEP 703 Future:** Python 3.13 introduces an experimental build flag to disable the GIL (No-GIL). With No-GIL, moving files using standard threads (like our `asyncio.to_thread`) could yield true hardware parallelism even for CPU-bound tasks in the future, dramatically changing how we architect asynchronous Python scripts.
