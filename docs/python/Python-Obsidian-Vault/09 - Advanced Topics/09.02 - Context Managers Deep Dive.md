## Creating Context Managers (Class-based)

```python
from types import TracebackType

class ManagedFile:
    """A custom context manager for file handling."""
    def __init__(self, filename: str, mode: str) -> None:
        self.filename: str = filename
        self.mode: str = mode
        self.file = None

    def __enter__(self):
        self.file = open(self.filename, self.mode)
        return self.file

    def __exit__(self, exc_type: type[BaseException] | None, exc_val: BaseException | None, exc_tb: TracebackType | None) -> bool | None:
        if self.file:
            self.file.close()
        # Returning True suppresses the exception
        return False
```

## Context Manager as Decorator

```python
from collections.abc import Generator
from contextlib import contextmanager

@contextmanager
def managed_list() -> Generator[list[int], None, None]:
    """Context manager that yields a list and reports its length upon exit."""
    data: list[int] = []
    try:
        yield data
    finally:
        print(f"List had {len(data)} items")

with managed_list() as items:
    items.append(1)
    items.append(2)
# Prints: List had 2 items
```

## Asynchronous Context Managers

```python
import asyncio
from collections.abc import AsyncGenerator
from contextlib import asynccontextmanager

@asynccontextmanager
async def async_managed_resource() -> AsyncGenerator[str, None]:
    """Asynchronous context manager using contextlib."""
    print("Acquiring resource...")
    await asyncio.sleep(0.1)
    try:
        yield "AsyncResource"
    finally:
        print("Releasing resource...")
        await asyncio.sleep(0.1)

async def main() -> None:
    async with async_managed_resource() as res:
        print(f"Using {res}")

# asyncio.run(main())
```

## Nested Context Managers and ExitStack

```python
from contextlib import ExitStack

def process_files(filenames: list[str]) -> None:
    """Safely open multiple files dynamically using ExitStack."""
    with ExitStack() as stack:
        files = [
            stack.enter_context(open(fname, 'r'))
            for fname in filenames
        ]
        # All files are automatically closed when exiting the block
```

## Database Transaction Example

```python
from collections.abc import Generator
from contextlib import contextmanager
from typing import Any

class MockConnection:
    def commit(self) -> None: pass
    def rollback(self) -> None: pass

@contextmanager
def transaction(connection: MockConnection) -> Generator[MockConnection, None, None]:
    """Manages a database transaction, rolling back if an exception occurs."""
    try:
        yield connection
        connection.commit()
        print("Transaction committed")
    except Exception as e:
        connection.rollback()
        print(f"Transaction rolled back: {e}")
        raise
```

### Exercise 1: Memory Allocation Tracker
Using `tracemalloc` to monitor memory usage in a block of code.

```python
import tracemalloc
from collections.abc import Generator
from contextlib import contextmanager

@contextmanager
def memory_tracker(label: str = "Operation") -> Generator[None, None, None]:
    """Tracks memory allocation for a block of code."""
    tracemalloc.start()
    try:
        yield
    finally:
        current, peak = tracemalloc.get_traced_memory()
        print(f"[{label}] Memory used: {current / 10**6:.4f} MB; Peak: {peak / 10**6:.4f} MB")
        tracemalloc.stop()

with memory_tracker("List comprehension"):
    large_list: list[int] = [i * i for i in range(1000000)]
```

### Exercise 2: Timer Context Manager
```python
from collections.abc import Generator
from contextlib import contextmanager
import time

@contextmanager
def timer(label: str = "Operation") -> Generator[None, None, None]:
    """Times the execution of a code block."""
    start: float = time.perf_counter()
    yield
    elapsed: float = time.perf_counter() - start
    print(f"{label} took {elapsed:.4f} seconds")

with timer("Sorting"):
    sorted(range(1000000, 0, -1))
```
