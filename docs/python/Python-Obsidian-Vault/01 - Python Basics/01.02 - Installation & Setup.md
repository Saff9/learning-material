## Installing Python 3.12+

### Windows

1. Go to python.org and download the latest Python 3.12+ version.
2. **IMPORTANT**: Check "Add Python to PATH" during installation.
3. Verify installation:

```bash
python --version
# or
py --version
```

### macOS

Option 1 - Official Installer:
```bash
# Download from python.org or use Homebrew
brew install python
```

Option 2 - Using Homebrew:
```bash
brew install python@3.12
python3.12 --version
```

### Linux (Ubuntu/Debian)

```bash
sudo apt update
sudo apt install python3.12 python3-pip
python3.12 --version
```

## Understanding the Python Interpreter & The GIL

The standard Python interpreter (CPython) reads your code top-to-bottom, compiles it to bytecode, and executes it on the Python Virtual Machine. 

### The Global Interpreter Lock (GIL) and PEP 703
Historically, CPython uses a Global Interpreter Lock (GIL) that allows only one thread to execute Python bytecode at a time, preventing true parallelism in CPU-bound multi-threaded Python programs. However, **PEP 703** introduced a plan for making the GIL optional (Free-threading) in Python 3.13+, fundamentally evolving how Python handles parallel execution and memory management.

```python
import dis

def example_bytecode() -> int:
    """
    An example function to demonstrate bytecode compilation.
    """
    x: int = 1 + 2
    return x

if __name__ == "__main__":
    dis.dis(example_bytecode)
```

### Type Annotations and Custom Context Managers

Context managers safely manage resources (like files or locks). You can create them using classes (`__enter__` and `__exit__`) or via the `contextlib` module.

```python
from typing import Any, Generator
import time
from contextlib import contextmanager

class Timer:
    """
    A custom context manager using dunder methods to measure execution time.
    """
    def __enter__(self) -> 'Timer':
        self.start: float = time.perf_counter()
        return self

    def __exit__(self, exc_type: Any, exc_val: Any, exc_tb: Any) -> None:
        self.end: float = time.perf_counter()
        print(f"Class Timer: {self.end - self.start:.4f} seconds")

@contextmanager
def simple_timer() -> Generator[None, None, None]:
    """
    A custom context manager using contextlib to measure execution time.
    """
    start: float = time.perf_counter()
    try:
        yield
    finally:
        end: float = time.perf_counter()
        print(f"Generator Timer: {end - start:.4f} seconds")

def process_data(data: list[str]) -> dict[str, int]:
    """
    Processes a list of strings and returns a mapping of string to its length.
    """
    return {item: len(item) for item in data}

if __name__ == "__main__":
    with Timer():
        result: dict[str, int] = process_data(["Python", "3.12", "Setup"])
        print(result)
        
    with simple_timer():
        _ = process_data(["Context", "Managers", "Are", "Awesome"])
```

### Async/Await Deep Dive

Asynchronous programming (`asyncio`) is crucial for I/O-bound tasks, allowing concurrency on a single thread. The event loop manages `Task` objects, and `asyncio.gather` can await multiple coroutines concurrently.

```python
import asyncio

async def fetch_data(task_id: int, delay: float) -> str | None:
    """
    Simulates an asynchronous data fetch using asyncio.sleep.
    
    Args:
        task_id: The ID of the task.
        delay: Time to wait in seconds.
        
    Returns:
        A completion message or None.
    """
    print(f"Task {task_id}: Starting...")
    await asyncio.sleep(delay)
    return f"Task {task_id}: Completed in {delay}s!"

async def async_main() -> None:
    """
    Main async entry point to demonstrate the event loop, tasks, and gather.
    """
    # Create tasks
    tasks: list[asyncio.Task[str | None]] = [
        asyncio.create_task(fetch_data(1, 1.5)),
        asyncio.create_task(fetch_data(2, 0.5)),
        asyncio.create_task(fetch_data(3, 1.0))
    ]
    
    # Run them concurrently and gather results
    results: list[str | None] = await asyncio.gather(*tasks)
    
    for res in results:
        if res is not None:
            print(res)

if __name__ == "__main__":
    # asyncio.run handles starting the event loop and tearing it down
    asyncio.run(async_main())
```

### Advanced Standard Library Primitives
Python provides powerful functional and collection primitives.

```python
import itertools
import functools
import collections

# 1. collections: namedtuple and Counter
Point = collections.namedtuple('Point', ['x', 'y'])
def analyze_collections() -> None:
    """Demonstrates collections primitives."""
    p: Point = Point(10, 20)
    counts: collections.Counter[str] = collections.Counter(["apple", "banana", "apple"])
    print(f"Point: {p.x}, {p.y} | Counts: {counts}")

# 2. functools: lru_cache for memoization
@functools.lru_cache(maxsize=128)
def fibonacci(n: int) -> int:
    """Calculates Fibonacci numbers with memoization."""
    if n < 2:
        return n
    return fibonacci(n-1) + fibonacci(n-2)

# 3. itertools: combinations and cycling
def explore_itertools() -> None:
    """Demonstrates itertools primitives."""
    items: list[str] = ["A", "B", "C"]
    combos: list[tuple[str, ...]] = list(itertools.combinations(items, 2))
    print(f"Combinations: {combos}")

if __name__ == "__main__":
    analyze_collections()
    print(f"Fib(10): {fibonacci(10)}")
    explore_itertools()
```

### Memory Profiling with `tracemalloc`
For advanced performance analysis, Python's `tracemalloc` allows you to track memory allocations.

```python
import tracemalloc

def memory_intensive_task() -> list[str]:
    """Generates a large list to demonstrate memory profiling."""
    return [f"Item {i}" * 100 for i in range(100_000)]

def profile_memory() -> None:
    """Profiles memory usage of a specific task."""
    tracemalloc.start()
    
    _ = memory_intensive_task()
    
    current: int
    peak: int
    current, peak = tracemalloc.get_traced_memory()
    print(f"Current memory usage: {current / 10**6:.2f} MB")
    print(f"Peak memory usage: {peak / 10**6:.2f} MB")
    
    tracemalloc.stop()

if __name__ == "__main__":
    profile_memory()
```
