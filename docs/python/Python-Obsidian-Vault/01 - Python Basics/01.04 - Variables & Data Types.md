## Variables

Variables are containers for storing data values. In modern Python, using strict type hints improves code readability and tooling support.

```python
from typing import Any

def demonstrate_variables() -> None:
    """Demonstrates basic variable creation with strict type annotations."""
    # Creating variables
    name: str = "Alice"
    age: int = 25
    height: float = 1.75
    is_student: bool = True
    
    # Python is dynamically typed at runtime, but static types help tools
    x: Any = 5        # Initially an int
    x = "hello"       # Now it is a string

    print(f"User {name}, Age: {age}, Height: {height}, Student: {is_student}")

demonstrate_variables()
```

## 🚀 Advanced Topics in Python 3.12+

### Custom Context Managers

Context managers allow precise resource management. You can use classes with `__enter__` and `__exit__`, or the `@contextmanager` decorator.

```python
from contextlib import contextmanager
from typing import Generator, Any

class ManagedResource:
    """A context manager implemented as a class."""
    def __enter__(self) -> 'ManagedResource':
        print("Acquiring resource...")
        return self
        
    def __exit__(self, exc_type: Any, exc_val: Any, exc_tb: Any) -> None:
        print("Releasing resource...")

@contextmanager
def managed_resource_gen() -> Generator[str, None, None]:
    """A context manager implemented via a generator."""
    print("Setup resource...")
    yield "Resource ready"
    print("Cleanup resource...")

def run_context_managers() -> None:
    """Demonstrates usage of custom context managers."""
    with ManagedResource():
        print("Inside class context manager")
        
    with managed_resource_gen() as res:
        print(f"Inside generator context manager: {res}")

run_context_managers()
```

### Async/Await Deep Dive

Modern Python heavily relies on `asyncio` for non-blocking operations.

```python
import asyncio

async def fetch_data(task_id: int) -> str:
    """Simulates an asynchronous I/O operation."""
    print(f"Task {task_id}: starting...")
    await asyncio.sleep(1)
    print(f"Task {task_id}: finished.")
    return f"Result {task_id}"

async def main_async() -> None:
    """Runs multiple async tasks concurrently."""
    tasks: list[asyncio.Task[str]] = [
        asyncio.create_task(fetch_data(1)),
        asyncio.create_task(fetch_data(2)),
    ]
    results: list[str] = await asyncio.gather(*tasks)
    print(f"All results: {results}")

# To run:
# asyncio.run(main_async())
```

### Advanced Collections, `itertools`, and `functools`

```python
import itertools
import functools
from collections import defaultdict, Counter
from typing import DefaultDict

def advanced_collections() -> None:
    """Demonstrates advanced primitives."""
    # Collections
    counts: Counter[str] = Counter(["apple", "banana", "apple"])
    grouped: DefaultDict[str, list[int]] = defaultdict(list)
    grouped["a"].append(1)
    
    # Itertools
    combinations: list[tuple[int, ...]] = list(itertools.combinations([1, 2, 3], 2))
    
    print(f"Counts: {counts}")
    print(f"Combinations: {combinations}")

@functools.lru_cache(maxsize=128)
def fibonacci(n: int) -> int:
    """Calculates Fibonacci numbers with memoization."""
    if n < 2:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

advanced_collections()
print(f"Fib(10) = {fibonacci(10)}")
```

### Memory Profiling & PEP 703 (Free-Threading)

```python
import tracemalloc

def memory_profile() -> None:
    """Demonstrates memory profiling with tracemalloc."""
    tracemalloc.start()
    
    # Allocate some memory
    large_list: list[int] = [i for i in range(100000)]
    
    current, peak = tracemalloc.get_traced_memory()
    print(f"Current memory usage is {current / 10**6}MB; Peak was {peak / 10**6}MB")
    tracemalloc.stop()

memory_profile()
```

> **Note on PEP 703 (Making the Global Interpreter Lock Optional):**
> Python 3.13+ introduces experimental support for free-threading (no GIL). This allows true multithreading in pure Python without needing multiprocessing. When utilizing these new builds, threads can execute Python bytecodes concurrently on multiple cores.
