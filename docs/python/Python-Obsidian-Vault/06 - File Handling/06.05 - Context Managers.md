## The `with` Statement

```python
# Without context manager (Prone to resource leaks if errors occur)
def legacy_read() -> str:
    file = open("data.txt", "r")
    try:
        return file.read()
    finally:
        file.close()

# With context manager (Safe, clean, guaranteed cleanup)
def modern_read() -> str:
    with open("data.txt", "r", encoding="utf-8") as file:
        return file.read()
```

## Class-Based Context Managers

A class must implement `__enter__` and `__exit__` to act as a context manager.

```python
from typing import Any, Type
from types import TracebackType

class FileManager:
    """A custom file manager showcasing __enter__ and __exit__."""
    def __init__(self, filename: str, mode: str) -> None:
        self.filename = filename
        self.mode = mode
        self.file: Any = None

    def __enter__(self) -> Any:
        self.file = open(self.filename, self.mode, encoding="utf-8")
        return self.file

    def __exit__(self, exc_type: Type[BaseException] | None, exc_val: BaseException | None, exc_tb: TracebackType | None) -> bool:
        if self.file:
            self.file.close()
        # Returning True suppresses exceptions
        return False
```

## Function-Based Context Managers

You can use the `@contextmanager` decorator from `contextlib` to create context managers using generators.

```python
from contextlib import contextmanager
from typing import Iterator

@contextmanager
def managed_resource(name: str) -> Iterator[dict[str, str]]:
    """Yields a resource dictionary and ensures cleanup."""
    print(f"Acquiring {name}")
    resource = {"name": name}
    try:
        yield resource
    finally:
        print(f"Releasing {name}")

def use_resource() -> None:
    with managed_resource("database") as db:
        print(f"Using {db['name']}")
```

## Asynchronous Context Managers

Modern Python uses `async with` for async resources (like network connections or async files).

```python
import asyncio
from contextlib import asynccontextmanager
from typing import AsyncIterator

@asynccontextmanager
async def async_database_connection(db_url: str) -> AsyncIterator[dict[str, str]]:
    """Simulates an asynchronous database connection manager."""
    print(f"Connecting asynchronously to {db_url}...")
    await asyncio.sleep(0.5)  # Simulate network I/O
    conn = {"url": db_url, "status": "connected"}
    try:
        yield conn
    finally:
        print(f"Closing async connection to {db_url}...")
        await asyncio.sleep(0.1)

async def test_async_cm() -> None:
    async with async_database_connection("postgresql://localhost") as db:
        print(f"Connected: {db['status']}")
```

## Multiple Context Managers

```python
def process_multiple_files(in_path: str, out_path: str) -> None:
    """Opens multiple context managers simultaneously."""
    with open(in_path, "r", encoding="utf-8") as infile, \\
         open(out_path, "w", encoding="utf-8") as outfile:
        for line in infile:
            outfile.write(line.upper())
```

### Exercise 1: Temporary Directory
```python
from contextlib import contextmanager
import tempfile
import shutil
from typing import Iterator

@contextmanager
def temp_directory() -> Iterator[str]:
    """Creates a temporary directory that is automatically deleted."""
    temp_dir: str = tempfile.mkdtemp()
    try:
        yield temp_dir
    finally:
        shutil.rmtree(temp_dir)
```

### Exercise 2: Execution Timer
```python
from contextlib import contextmanager
import time
from typing import Iterator

@contextmanager
def timer(operation_name: str) -> Iterator[None]:
    """Times an operation block."""
    start: float = time.perf_counter()
    yield
    elapsed: float = time.perf_counter() - start
    print(f"[{operation_name}] Elapsed: {elapsed:.4f}s")

with timer("Looping"):
    sum(range(1_000_000))
```
