## Basic File Writing

```python
from typing import Any

def write_basic_file(filepath: str, content: str) -> None:
    """
    Writes content to a file, overwriting if it exists.
    
    Args:
        filepath (str): The destination file path.
        content (str): The text to write.
    """
    # Write mode (overwrites existing file)
    with open(filepath, "w", encoding="utf-8") as f:
        f.write(content)

def append_to_file(filepath: str, content: str) -> None:
    """Appends content to the end of a file."""
    # Append mode (adds to existing file)
    with open(filepath, "a", encoding="utf-8") as f:
        f.write(content + "\\n")
```

## Write Modes

```python
def demonstrate_write_modes() -> None:
    """Demonstrates various file writing modes."""
    # "x" - Exclusive creation (fails if exists)
    try:
        with open("must_be_new.txt", "x", encoding="utf-8") as f:
            f.write("First and only")
    except FileExistsError:
        print("File already exists!")

    # "wb" - Write binary
    with open("data.bin", "wb") as f:
        binary_data: bytes = b"\\x00\\x01\\x02"
        f.write(binary_data)
```

## Writing Multiple Lines

```python
def write_multiple_lines(filepath: str, lines: list[str]) -> None:
    """
    Writes a list of strings to a file efficiently.
    """
    # Using writelines generator
    with open(filepath, "w", encoding="utf-8") as f:
        f.writelines(line + "\\n" for line in lines)

def write_with_print(filepath: str, lines: list[str]) -> None:
    """Uses print to write lines, which adds newlines automatically."""
    with open(filepath, "w", encoding="utf-8") as f:
        for line in lines:
            print(line, file=f)
```

### Exercise 1: Logger Function

```python
from datetime import datetime

def log_message(filename: str, message: str) -> None:
    """Logs a message with a timestamp."""
    timestamp: str = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    with open(filename, "a", encoding="utf-8") as f:
        f.write(f"[{timestamp}] {message}\\n")

log_message("app.log", "Application started")
```

### Exercise 2: Save and Load Data
```python
def save_numbers(filename: str, numbers: list[int]) -> None:
    """Saves a list of integers to a file."""
    with open(filename, "w", encoding="utf-8") as f:
        for num in numbers:
            f.write(f"{num}\\n")

def load_numbers(filename: str) -> list[int]:
    """Loads integers from a file."""
    with open(filename, "r", encoding="utf-8") as f:
        return [int(line.strip()) for line in f]

save_numbers("numbers.txt", [1, 2, 3, 4, 5])
```

### Exercise 3: File Buffering (Advanced)
```python
def write_large_buffered(filename: str, data_chunks: list[str]) -> None:
    """Writes data with a specific buffer size (e.g., 8192 bytes)."""
    with open(filename, "w", buffering=8192, encoding="utf-8") as f:
        for chunk in data_chunks:
            f.write(chunk)
```
