## Basic File Reading

```python
from typing import Any
import tracemalloc

def read_basic_file(filepath: str) -> str:
    """
    Reads the entire content of a file using a context manager.
    
    Args:
        filepath (str): The path to the file.
        
    Returns:
        str: The contents of the file.
    """
    # Using with statement (recommended)
    with open(filepath, "r", encoding="utf-8") as file:
        content: str = file.read()
        return content

# Without with (must close manually - not recommended)
def legacy_read(filepath: str) -> str:
    """Legacy approach to reading files (requires manual cleanup)."""
    file = open(filepath, "r", encoding="utf-8")
    try:
        return file.read()
    finally:
        file.close()
```

## Read Modes & Methods

```python
def read_modes_example() -> None:
    """Demonstrates various file reading modes."""
    # "r" - Read (default)
    with open("file.txt", "r", encoding="utf-8") as f:
        data: str = f.read()

    # "rb" - Read binary
    with open("image.png", "rb") as f:
        binary_data: bytes = f.read()

    # "r+" - Read and write
    with open("file.txt", "r+", encoding="utf-8") as f:
        data = f.read()
        f.write("new data")
```

## Reading Line by Line (Memory Efficient)

Using `tracemalloc` to prove memory efficiency when using generators vs reading all at once.

```python
import tracemalloc
from typing import Iterator

def read_large_file_generator(filepath: str) -> Iterator[str]:
    """
    Yields lines from a file one by one.
    Memory efficient for large files.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        for line in f:
            yield line.strip()

def analyze_memory(filepath: str) -> None:
    """Analyzes memory usage of file reading."""
    tracemalloc.start()
    
    # Efficient iteration
    for _ in read_large_file_generator(filepath):
        pass
        
    current, peak = tracemalloc.get_traced_memory()
    print(f"Current memory usage is {current / 10**6}MB; Peak was {peak / 10**6}MB")
    tracemalloc.stop()
```

## Checking File Existence

```python
import os
from pathlib import Path

def check_and_read(filepath: str | Path) -> str | None:
    """Safely checks if a file exists before reading."""
    path = Path(filepath) if isinstance(filepath, str) else filepath
    
    if path.exists():
        return path.read_text(encoding="utf-8")
    return None
```

### Exercise 1: Word Counter
```python
def count_words(filepath: str) -> dict[str, int]:
    """Counts total words and lines in a file."""
    with open(filepath, "r", encoding="utf-8") as f:
        text: str = f.read()
        words: list[str] = text.split()
        return {
            "total_words": len(words),
            "total_lines": text.count(chr(10)) + 1
        }
```

### Exercise 2: Search in File
```python
def search_file(filename: str, keyword: str) -> list[tuple[int, str]]:
    """Searches for a keyword in a file, returning line numbers and content."""
    matches: list[tuple[int, str]] = []
    with open(filename, "r", encoding="utf-8") as f:
        for line_num, line in enumerate(f, 1):
            if keyword in line:
                matches.append((line_num, line.strip()))
    return matches
```
