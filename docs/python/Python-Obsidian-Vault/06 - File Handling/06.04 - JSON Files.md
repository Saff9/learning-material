## Reading JSON

```python
import json
from typing import Any

def read_json_string(json_string: str) -> dict[str, Any]:
    """
    Parses a JSON string into a Python dictionary.
    """
    data: dict[str, Any] = json.loads(json_string)
    return data

def read_json_file(filepath: str) -> Any:
    """
    Reads JSON data from a file.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        data: Any = json.load(f)
        return data
```

## Writing JSON

```python
import json
from typing import Any

def write_json_file(filepath: str, data: dict[str, Any] | list[Any]) -> None:
    """
    Writes Python data structures to a JSON file.
    """
    with open(filepath, "w", encoding="utf-8") as f:
        # indent=2 makes the output pretty-printed
        json.dump(data, f, indent=2, sort_keys=True)
```

## Type Conversions

| Python | JSON | Strict Type Hint |
|--------|------|-----------------|
| dict | object | `dict[str, Any]` |
| list, tuple | array | `list[Any]` |
| str | string | `str` |
| int, float | number | `int | float` |
| True | true | `bool` |
| False | false | `bool` |
| None | null | `None` |

## Handling Custom Objects (Encoders)

Standard JSON encoders do not support complex types like `datetime` or custom classes.

```python
import json
from datetime import datetime
from typing import Any

class DateTimeEncoder(json.JSONEncoder):
    """Custom JSON Encoder that handles datetime objects."""
    def default(self, obj: Any) -> Any:
        if isinstance(obj, datetime):
            return obj.isoformat()
        return super().default(obj)

def encode_complex_data() -> str:
    """Encodes a dictionary containing a datetime object."""
    data: dict[str, Any] = {"created": datetime.now()}
    return json.dumps(data, cls=DateTimeEncoder)
```

## Asynchronous JSON Loading (Advanced)

For extremely large JSON payloads in async contexts, blocking the event loop is a risk. Consider `asyncio.to_thread` for CPU-bound parsing.

```python
import asyncio
import json
from typing import Any

async def load_large_json_async(json_string: str) -> Any:
    """
    Parses a very large JSON string asynchronously to avoid blocking
    the asyncio event loop.
    """
    # json.loads is CPU-bound, so we offload it to a thread
    data = await asyncio.to_thread(json.loads, json_string)
    return data
```

## 🧪 Practice

### Exercise 1: Config File Manager
```python
import json
from typing import Any

def load_config(filename: str) -> dict[str, Any]:
    """Loads configuration settings from JSON."""
    with open(filename, "r", encoding="utf-8") as f:
        return json.load(f)

def save_config(filename: str, config: dict[str, Any]) -> None:
    """Saves configuration settings to JSON."""
    with open(filename, "w", encoding="utf-8") as f:
        json.dump(config, f, indent=2)
```

### Exercise 2: Merge JSON Files
```python
import json
from typing import Any

def merge_json_files(files: list[str], output: str) -> None:
    """Merges multiple JSON files into a single file."""
    merged: dict[str, Any] = {}
    for filename in files:
        with open(filename, "r", encoding="utf-8") as f:
            data: dict[str, Any] = json.load(f)
            merged.update(data)

    with open(output, "w", encoding="utf-8") as f:
        json.dump(merged, f, indent=2)
```
