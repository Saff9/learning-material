## 🏗️ String Memory & PEP 703 (Free-Threading)

Strings in Python are immutable arrays of Unicode characters. In Python 3.12+, understanding string memory allocation, interning, and how the Global Interpreter Lock (GIL) impacts them is crucial.

### String Interning and Memory Profiling
Python optimizes memory by "interning" small or commonly used strings (like identifiers). With PEP 703 (making the GIL optional), immutable objects like strings are inherently thread-safe to share across threads without locking, massively boosting parallel text processing performance.

```python
import sys
from typing import Final

def memory_profile_demo() -> None:
    """Demonstrate string interning and memory usage."""
    # Interned strings (same object in memory)
    s1: Final[str] = "hello_world"
    s2: Final[str] = "hello_world"
    print(f"Are s1 and s2 the same object? {s1 is s2}")  # True
    
    # Dynamically created strings might not be interned automatically
    s3: str = "".join(["hello", "_", "world"])
    print(f"Are s1 and s3 the same object? {s1 is s3}")  # False
    
    # You can manually intern strings if memory optimization is critical
    import sys
    s4: str = sys.intern(s3)
    print(f"Are s1 and s4 the same object? {s1 is s4}")  # True
    print(f"Memory size of s1: {sys.getsizeof(s1)} bytes")

if __name__ == "__main__":
    memory_profile_demo()
```

## ⚡ Context Managers for String Operations

Context managers are excellent for maintaining state during complex text transformations or handling streaming text securely.

```python
from types import TracebackType
from typing import Optional, Type, TextIO
import io

class TextProcessorContext:
    """A context manager to safely build and sanitize a large string buffer."""
    
    def __init__(self, initial_text: str = "") -> None:
        self._buffer: io.StringIO = io.StringIO(initial_text)
    
    def __enter__(self) -> io.StringIO:
        return self._buffer
        
    def __exit__(
        self,
        exc_type: Optional[Type[BaseException]],
        exc_val: Optional[BaseException],
        exc_tb: Optional[TracebackType]
    ) -> bool:
        if exc_type is not None:
            print(f"Error during string processing: {exc_val}")
        self._buffer.close()
        return False

# Usage:
def process_logs() -> None:
    with TextProcessorContext() as text_buffer:
        text_buffer.write("LOG: System started.\n")
        text_buffer.write("LOG: Processing items...")
        print(text_buffer.getvalue())

if __name__ == "__main__":
    process_logs()
```

## 🔄 High-Performance String Tools (`itertools`, `functools`, `collections`)

Using native looping for strings is often slow. Modern Python leverages built-in C-optimized modules.

```python
from collections import Counter, deque
from itertools import groupby, permutations
from functools import reduce
from typing import List, Dict, Iterator

def string_advanced_tools(text: str) -> None:
    """Demonstrate advanced text processing using standard libraries."""
    
    # 1. collections.Counter for character frequency
    freq: Counter[str] = Counter(text)
    print(f"Most common chars in '{text}': {freq.most_common(2)}")
    
    # 2. itertools.groupby for consecutive duplicate removal
    # e.g., 'aaabbc' -> 'abc'
    sorted_text: str = "".join(sorted(text))
    deduped: str = "".join(k for k, _ in groupby(text))
    print(f"Deduped consecutive chars: {deduped}")
    
    # 3. functools.reduce for complex string aggregations
    words: List[str] = ["Python", "3.12", "is", "fast"]
    sentence: str = reduce(lambda acc, w: f"{acc} {w}", words).strip()
    print(f"Reduced sentence: {sentence}")

if __name__ == "__main__":
    string_advanced_tools("mississippi")
```

## 🚀 Asynchronous String Processing (async/await)

When dealing with I/O bound string operations (e.g., fetching text from APIs, processing massive log files), `async/await` prevents blocking the event loop.

```python
import asyncio
from typing import AsyncIterator

async def async_text_streamer(text: str, chunk_size: int = 3) -> AsyncIterator[str]:
    """Asynchronously stream chunks of a string."""
    for i in range(0, len(text), chunk_size):
        await asyncio.sleep(0.1)  # Simulate I/O bound delay
        yield text[i:i + chunk_size]

async def process_async_text() -> None:
    """Consume the asynchronous text stream."""
    sample_text: str = "Modern Python String Processing"
    result_buffer: list[str] = []
    
    async for chunk in async_text_streamer(sample_text):
        result_buffer.append(chunk.upper())
        
    final_string: str = "".join(result_buffer)
    print(f"Async Processed: {final_string}")

if __name__ == "__main__":
    asyncio.run(process_async_text())
```

## 🛠️ Modern String Formatting (f-strings 3.12+)

Python 3.12 introduced quote reuse, multi-line expressions, and backslashes inside f-strings!

```python
from typing import List

def modern_fstrings_demo() -> None:
    """Showcase Python 3.12+ f-string capabilities."""
    words: List[str] = ['Python', '3.12', 'Strings']
    
    # 3.12 allows reusing quotes inside the f-string!
    joined_str: str = f"Joined: {', '.join(words)}"
    
    # 3.12 allows backslashes inside expressions!
    newline_str: str = f"With Newlines: {'\n'.join(words)}"
    
    print(joined_str)
    print(newline_str)

if __name__ == "__main__":
    modern_fstrings_demo()
```

### Exercise 1: Strict Typed Palindrome Checker
```python
import re

def is_palindrome(s: str) -> bool:
    """Check if a string is a palindrome, ignoring non-alphanumeric characters."""
    cleaned: str = re.sub(r'[^A-Za-z0-9]', '', s).lower()
    return cleaned == cleaned[::-1]

if __name__ == "__main__":
    print(f"Is palindrome? {is_palindrome('A man, a plan, a canal: Panama')}")  # True
```

### Exercise 2: Async File Regex Parsing
```python
import asyncio
import re
from typing import List

async def extract_emails_async(text: str) -> List[str]:
    """Asynchronously extract emails from a text block."""
    await asyncio.sleep(0.01) # Simulate yielding control
    email_pattern: re.Pattern[str] = re.compile(r'[\w\.-]+@[\w\.-]+\.\w+')
    return email_pattern.findall(text)

async def run_extraction() -> None:
    text: str = "Contact: john@email.com or jane@site.org"
    emails: List[str] = await extract_emails_async(text)
    print(f"Found emails: {emails}")

if __name__ == "__main__":
    asyncio.run(run_extraction())
```
