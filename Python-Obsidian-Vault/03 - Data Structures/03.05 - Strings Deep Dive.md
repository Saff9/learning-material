## String Creation and Properties

```python
# Multiple ways
s1 = "Hello"
s2 = 'World'
s3 = """Multi
line"""
s4 = str(42)

# Immutability
s = "hello"
# s[0] = "H"  # ERROR!

# But you can create new strings
s = "H" + s[1:]  # "Hello"
```

## Essential String Methods

```python
text = "  Hello, World!  "

# Case
print(text.lower())      # "  hello, world!  "
print(text.upper())      # "  HELLO, WORLD!  "
print(text.title())      # "  Hello, World!  "
print(text.swapcase())   # "  hELLO, wORLD!  "

# Strip whitespace
print(text.strip())      # "Hello, World!"
print(text.lstrip())     # "Hello, World!  "
print(text.rstrip())     # "  Hello, World!"

# Search
print(text.find("World"))     # 9 (or -1 if not found)
print(text.index("World"))    # 9 (raises ValueError if not found)
print("World" in text)        # True

# Replace
print(text.replace("World", "Python"))  # "  Hello, Python!  "

# Split and Join
words = "a,b,c".split(",")    # ['a', 'b', 'c']
joined = "-".join(words)      # "a-b-c"
```

## String Checking Methods

```python
print("123".isdigit())      # True
print("abc".isalpha())      # True
print("abc123".isalnum())   # True
print("   ".isspace())      # True
print("Hello".istitle())    # True
print("hello".islower())    # True
print("HELLO".isupper())    # True

# Start/End
print("hello.txt".startswith("hello"))  # True
print("hello.txt".endswith(".txt"))   # True
```

## String Formatting

```python
name = "Alice"
age = 25

# f-strings (recommended)
print(f"Name: {name}, Age: {age}")

# Format specifiers
pi = 3.14159
print(f"PI = {pi:.2f}")           # 3.14
print(f"Number: {42:05d}")        # 00042
print(f"Percent: {0.85:.1%}")     # 85.0%
print(f"Centered: {'hi':^10}")    # "    hi    "

# Alignment
print(f"{'Left':<10}|")   # "Left      |"
print(f"{'Right':>10}|")  # "     Right|"
print(f"{'Center':^10}|") # "  Center  |"
```

## Regular Expressions (re module)

```python
import re

text = "Contact: john@email.com or jane@site.org"

# Find all emails
emails = re.findall(r'\w+@\w+\.\w+', text)
print(emails)

# Match pattern
if re.match(r'\d{3}-\d{4}', "123-4567"):
    print("Valid format")

# Replace
new_text = re.sub(r'\d+', 'X', "Room 101, Floor 2")
print(new_text)  # "Room X, Floor X"

# Split
parts = re.split(r'\s+', "a  b   c")
print(parts)  # ['a', 'b', 'c']
```

def is_palindrome(s):
    cleaned = ''.join(c.lower() for c in s if c.isalnum())
    return cleaned == cleaned[::-1]

print(is_palindrome("A man, a plan, a canal: Panama"))  # True

### Exercise 3: Caesar Cipher
```python
def caesar_cipher(text, shift):
    result = ""
    for char in text:
        if char.isalpha():
            base = ord('A') if char.isupper() else ord('a')
            result += chr((ord(char) - base + shift) % 26 + base)
        else:
            result += char
    return result

print(caesar_cipher("Hello", 3))  # "Khoor"
```

### Exercise 4: Word Wrap
```python
def word_wrap(text, width):
    words = text.split()
    lines = []
    current = []
    current_len = 0

    for word in words:
        if current_len + len(word) + len(current) > width:
            lines.append(" ".join(current))
            current = [word]
            current_len = len(word)
        else:
            current.append(word)
            current_len += len(word)

    if current:
        lines.append(" ".join(current))

    return "\n".join(lines)
```
