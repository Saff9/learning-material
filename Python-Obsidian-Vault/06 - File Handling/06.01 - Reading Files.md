## Basic File Reading

```python
# Using with statement (recommended)
with open("data.txt", "r") as file:
    content = file.read()
    print(content)

# Without with (must close manually)
file = open("data.txt", "r")
content = file.read()
file.close()
```

## Read Modes

```python
# "r" - Read (default)
with open("file.txt", "r") as f:
    data = f.read()

# "rb" - Read binary
with open("image.png", "rb") as f:
    data = f.read()

# "r+" - Read and write
with open("file.txt", "r+") as f:
    data = f.read()
    f.write("new data")
```

## Reading Methods

```python
with open("data.txt", "r") as f:
    # Read all at once
    all_content = f.read()

    # Read specific size
    f.seek(0)  # Go back to start
    chunk = f.read(100)  # First 100 chars

    # Read one line
    f.seek(0)
    first_line = f.readline()

    # Read all lines into list
    f.seek(0)
    lines = f.readlines()
```

## Reading Line by Line (Memory Efficient)

```python
# Best for large files
with open("large_file.txt", "r") as f:
    for line in f:
        print(line.strip())

# With line numbers
with open("data.txt", "r") as f:
    for line_num, line in enumerate(f, 1):
        print(f"{line_num}: {line.strip()}")
```

## Checking File Existence

```python
import os

if os.path.exists("data.txt"):
    with open("data.txt", "r") as f:
        content = f.read()
else:
    print("File not found!")

# Or use try-except
try:
    with open("data.txt", "r") as f:
        content = f.read()
except FileNotFoundError:
    print("File not found!")
```

### Exercise 2: Search in File
```python
def search_file(filename, keyword):
    matches = []
    with open(filename, "r") as f:
        for line_num, line in enumerate(f, 1):
            if keyword in line:
                matches.append((line_num, line.strip()))
    return matches

results = search_file("data.txt", "Python")
for line_num, line in results:
    print(f"Line {line_num}: {line}")
```

### Exercise 3: File Statistics
```python
def file_stats(filename):
    with open(filename, "r") as f:
        content = f.read()

    return {
        "chars": len(content),
        "words": len(content.split()),
        "lines": content.count("\n") + 1,
        "bytes": len(content.encode("utf-8"))
    }

stats = file_stats("sample.txt")
for key, value in stats.items():
    print(f"{key}: {value}")
```
