## Creating Paths

```python
from pathlib import Path

# Current directory
p = Path(".")

# Absolute path
home = Path.home()
documents = home / "Documents"

# From string
file_path = Path("/home/user/documents/file.txt")

# Building paths
base = Path("/home/user")
file = base / "documents" / "file.txt"
print(file)  # /home/user/documents/file.txt
```

## Path Properties

```python
p = Path("/home/user/documents/file.txt")

print(p.name)       # file.txt
print(p.stem)       # file
print(p.suffix)     # .txt
print(p.parent)     # /home/user/documents
print(p.parents[0]) # /home/user/documents
print(p.parents[1]) # /home/user
print(p.parts)      # ('', 'home', 'user', 'documents', 'file.txt')
print(p.absolute()) # Absolute path
```

## Checking Files

```python
p = Path("data.txt")

print(p.exists())      # True/False
print(p.is_file())     # True/False
print(p.is_dir())      # True/False
print(p.is_absolute()) # True/False
```

## File Operations

```python
p = Path("example.txt")

# Read
content = p.read_text()
bytes_data = p.read_bytes()

# Write
p.write_text("Hello, World!")
p.write_bytes(b"\x00\x01\x02")

# Rename
p.rename("new_name.txt")

# Delete
p.unlink()  # Remove file
p.rmdir()   # Remove empty directory
```

## Directory Operations

```python
# List directory
for item in Path(".").iterdir():
    print(item)

# Glob patterns
for py_file in Path(".").glob("*.py"):
    print(py_file)

# Recursive glob
for file in Path(".").rglob("*.txt"):
    print(file)

# Create directory
new_dir = Path("new_folder")
new_dir.mkdir(exist_ok=True)
new_dir.mkdir(parents=True, exist_ok=True)  # Create parents too
```

### Exercise 1: File Organizer
```python
from pathlib import Path
import shutil

def organize_by_extension(directory):
    path = Path(directory)
    for file in path.iterdir():
        if file.is_file():
            ext = file.suffix[1:] or "no_extension"
            folder = path / ext
            folder.mkdir(exist_ok=True)
            shutil.move(str(file), str(folder / file.name))

organize_by_extension("downloads")
```

### Exercise 2: Find Large Files
```python
from pathlib import Path

def find_large_files(directory, min_size_mb=10):
    path = Path(directory)
    min_bytes = min_size_mb * 1024 * 1024

    for file in path.rglob("*"):
        if file.is_file() and file.stat().st_size > min_bytes:
            size_mb = file.stat().st_size / (1024 * 1024)
            print(f"{file}: {size_mb:.1f} MB")

find_large_files(".", 1)
```

### Exercise 3: Backup Script
```python
from pathlib import Path
import shutil
from datetime import datetime

def backup_directory(source, backup_location):
    source_path = Path(source)
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    backup_name = f"{source_path.name}_{timestamp}"
    backup_path = Path(backup_location) / backup_name

    shutil.copytree(source_path, backup_path)
    print(f"Backup created: {backup_path}")

backup_directory("important_data", "backups")
```
