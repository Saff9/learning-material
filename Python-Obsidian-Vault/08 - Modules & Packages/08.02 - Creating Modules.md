## Creating a Module

A module is simply a `.py` file. Create `mymath.py`:

```python
# mymath.py
"""A simple math utility module."""

PI = 3.14159

def add(a, b):
    """Add two numbers."""
    return a + b

def multiply(a, b):
    """Multiply two numbers."""
    return a * b

class Calculator:
    """A simple calculator class."""

    def __init__(self):
        self.history = []

    def calculate(self, a, b, operation):
        if operation == "add":
            result = add(a, b)
        elif operation == "multiply":
            result = multiply(a, b)
        else:
            raise ValueError("Unknown operation")

        self.history.append(f"{a} {operation} {b} = {result}")
        return result

# This code only runs when the file is executed directly
if __name__ == "__main__":
    print("Running mymath module directly")
    calc = Calculator()
    print(calc.calculate(5, 3, "add"))
```

## Using the Module

```python
# main.py
import mymath

print(mymath.PI)                    # 3.14159
print(mymath.add(2, 3))           # 5
calc = mymath.Calculator()
print(calc.calculate(4, 5, "multiply"))  # 20
```

## Module Attributes

```python
import mymath

print(mymath.__name__)      # mymath
print(mymath.__file__)      # Path to the file
print(mymath.__doc__)       # Module docstring
print(dir(mymath))          # All names in module
```

## The `__all__` Variable

```python
# mymath.py
__all__ = ["add", "multiply", "Calculator"]
# Only these are imported with: from mymath import *

PI = 3.14159  # Not in __all__, so not imported with *
```

## Module Reloading

```python
import importlib
import mymath

# After modifying mymath.py
importlib.reload(mymath)
```

### Exercise 1: Create a String Utilities Module
```python
# string_utils.py
"""String utility functions."""

def reverse(s):
    """Reverse a string."""
    return s[::-1]

def count_vowels(s):
    """Count vowels in a string."""
    return sum(1 for c in s.lower() if c in "aeiou")

def is_palindrome(s):
    """Check if string is a palindrome."""
    cleaned = ''.join(c.lower() for c in s if c.isalnum())
    return cleaned == cleaned[::-1]

if __name__ == "__main__":
    print(reverse("hello"))
    print(count_vowels("hello"))
    print(is_palindrome("A man a plan a canal Panama"))
```

### Exercise 2: Create a Date Utilities Module
```python
# date_utils.py
from datetime import datetime, timedelta

def days_until(date_string):
    """Calculate days until a given date."""
    target = datetime.strptime(date_string, "%Y-%m-%d")
    return (target - datetime.now()).days

def is_weekend(date=None):
    """Check if a date is a weekend."""
    if date is None:
        date = datetime.now()
    return date.weekday() >= 5

def add_days(date_string, days):
    """Add days to a date."""
    date = datetime.strptime(date_string, "%Y-%m-%d")
    result = date + timedelta(days=days)
    return result.strftime("%Y-%m-%d")
```
