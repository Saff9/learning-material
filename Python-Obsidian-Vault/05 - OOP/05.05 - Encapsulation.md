## Access Modifiers in Python

Python uses naming conventions rather than strict access modifiers:

```python
class BankAccount:
    def __init__(self, owner, balance):
        self.owner = owner          # Public
        self._balance = balance      # Protected (convention)
        self.__pin = "1234"          # Private (name mangling)
```

## Public Attributes

```python
class Car:
    def __init__(self, make, model):
        self.make = make      # Public
        self.model = model    # Public
        self.year = 2024      # Public

car = Car("Toyota", "Camry")
print(car.make)     # OK
print(car.model)    # OK
car.year = 2025     # OK
```

## Protected Attributes (Convention)

```python
class Employee:
    def __init__(self, name, salary):
        self.name = name
        self._salary = salary  # Protected - should not access directly

    def get_salary(self):
        return self._salary

emp = Employee("Alice", 50000)
print(emp._salary)  # Works but discouraged
```

## Private Attributes (Name Mangling)

```python
class SecretKeeper:
    def __init__(self):
        self.__secret = "Top Secret"  # Private

    def reveal(self):
        return self.__secret

sk = SecretKeeper()
# print(sk.__secret)  # AttributeError!
print(sk.reveal())     # Works!

# Name mangling allows access (but don't do this!)
print(sk._SecretKeeper__secret)  # Top Secret
```

## Getters and Setters

```python
class Temperature:
    def __init__(self, celsius):
        self._celsius = celsius

    # Getter
    def get_celsius(self):
        return self._celsius

    # Setter with validation
    def set_celsius(self, value):
        if value < -273.15:
            raise ValueError("Below absolute zero!")
        self._celsius = value

    # Property (better way - see next note)
    celsius = property(get_celsius, set_celsius)

temp = Temperature(25)
print(temp.celsius)  # 25
temp.celsius = 30    # Uses setter
# temp.celsius = -300  # Raises ValueError
```

## 🧪 Practice

### Exercise 1: Secure Bank Account
```python
class SecureAccount:
    def __init__(self, owner, initial_balance):
        self.owner = owner
        self.__balance = initial_balance
        self.__transaction_history = []

    def deposit(self, amount):
        if amount > 0:
            self.__balance += amount
            self.__transaction_history.append(f"+${amount}")
            return True
        return False

    def withdraw(self, amount):
        if 0 < amount <= self.__balance:
            self.__balance -= amount
            self.__transaction_history.append(f"-${amount}")
            return True
        return False

    def get_balance(self):
        return self.__balance

    def get_history(self):
        return self.__transaction_history.copy()

account = SecureAccount("Alice", 1000)
account.deposit(500)
account.withdraw(200)
print(account.get_balance())
print(account.get_history())
```

### Exercise 2: Validated Person
```python
class Person:
    def __init__(self, name, age):
        self.__name = name
        self.__age = None
        self.set_age(age)

    def get_name(self):
        return self.__name

    def get_age(self):
        return self.__age

    def set_age(self, age):
        if not isinstance(age, int) or age < 0 or age > 150:
            raise ValueError("Invalid age")
        self.__age = age

person = Person("Alice", 25)
print(person.get_name())
# person.set_age(200)  # ValueError!
```
