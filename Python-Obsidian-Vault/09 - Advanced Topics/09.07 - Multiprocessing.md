## Basic Multiprocessing

```python
import multiprocessing
import time

def worker(name, delay):
    print(f"Process {name} starting (PID: {multiprocessing.current_process().pid})")
    time.sleep(delay)
    print(f"Process {name} done")

if __name__ == "__main__":
    process1 = multiprocessing.Process(target=worker, args=("A", 2))
    process2 = multiprocessing.Process(target=worker, args=("B", 1))

    process1.start()
    process2.start()

    process1.join()
    process2.join()

    print("All processes complete")
```

## Process Pool

```python
from multiprocessing import Pool
import time

def square(n):
    time.sleep(0.5)
    return n * n

if __name__ == "__main__":
    numbers = range(10)

    with Pool(processes=4) as pool:
        # Map
        results = pool.map(square, numbers)
        print(results)

        # Async
        result = pool.apply_async(square, (20,))
        print(result.get())
```

## Sharing Data

```python
from multiprocessing import Process, Value, Array
import time

def increment_counter(counter):
    for _ in range(1000):
        counter.value += 1

if __name__ == "__main__":
    counter = Value('i', 0)  # Shared integer

    processes = []
    for _ in range(4):
        p = Process(target=increment_counter, args=(counter,))
        processes.append(p)
        p.start()

    for p in processes:
        p.join()

    print(f"Counter: {counter.value}")
```

## Using Manager for Complex Data

```python
from multiprocessing import Process, Manager

def worker(d, l, name):
    d[name] = name * 2
    l.append(name)

if __name__ == "__main__":
    with Manager() as manager:
        shared_dict = manager.dict()
        shared_list = manager.list()

        processes = []
        for name in ["A", "B", "C"]:
            p = Process(target=worker, args=(shared_dict, shared_list, name))
            processes.append(p)
            p.start()

        for p in processes:
            p.join()

        print(dict(shared_dict))
        print(list(shared_list))
```

### Exercise 1: Parallel Computation
```python
from multiprocessing import Pool
import time

def is_prime(n):
    if n < 2:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

def find_primes(start, end):
    return [n for n in range(start, end) if is_prime(n)]

if __name__ == "__main__":
    ranges = [(2, 10000), (10000, 20000), (20000, 30000), (30000, 40000)]

    start_time = time.time()
    with Pool() as pool:
        results = pool.starmap(find_primes, ranges)

    all_primes = [p for sublist in results for p in sublist]
    print(f"Found {len(all_primes)} primes in {time.time() - start_time:.2f}s")
```

### Exercise 2: Parallel File Processing
```python
from multiprocessing import Pool
import os

def process_file(filename):
    if not os.path.exists(filename):
        return None
    with open(filename, 'r') as f:
        content = f.read()
    return {
        "filename": filename,
        "size": len(content),
        "lines": content.count("\n")
    }

if __name__ == "__main__":
    files = ["file1.txt", "file2.txt", "file3.txt"]

    with Pool() as pool:
        results = pool.map(process_file, files)

    for result in results:
        if result:
            print(f"{result['filename']}: {result['size']} bytes, {result['lines']} lines")
```
