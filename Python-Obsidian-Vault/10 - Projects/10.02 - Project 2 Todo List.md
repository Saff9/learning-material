```python
# todo.py
"""A command-line todo list application."""

import json
import os
from datetime import datetime

class TodoList:
    def __init__(self, filename="todos.json"):
        self.filename = filename
        self.tasks = []
        self.load()

    def load(self):
        """Load tasks from file."""
        if os.path.exists(self.filename):
            with open(self.filename, 'r') as f:
                self.tasks = json.load(f)

    def save(self):
        """Save tasks to file."""
        with open(self.filename, 'w') as f:
            json.dump(self.tasks, f, indent=2)

    def add(self, title, priority="medium"):
        """Add a new task."""
        task = {
            "id": len(self.tasks) + 1,
            "title": title,
            "priority": priority,
            "completed": False,
            "created": datetime.now().isoformat()
        }
        self.tasks.append(task)
        self.save()
        print(f"Added: {title}")

    def list_tasks(self, show_all=False):
        """List all tasks."""
        if not self.tasks:
            print("No tasks found!")
            return

        print("\n=== Your Tasks ===")
        for task in self.tasks:
            if not show_all and task["completed"]:
                continue
            status = "[X]" if task["completed"] else "[ ]"
            print(f"{status} {task['id']}. {task['title']} ({task['priority']})")

    def complete(self, task_id):
        """Mark a task as completed."""
        for task in self.tasks:
            if task["id"] == task_id:
                task["completed"] = True
                self.save()
                print(f"Completed: {task['title']}")
                return
        print("Task not found!")

    def delete(self, task_id):
        """Delete a task."""
        for i, task in enumerate(self.tasks):
            if task["id"] == task_id:
                removed = self.tasks.pop(i)
                self.save()
                print(f"Deleted: {removed['title']}")
                return
        print("Task not found!")

def main():
    todo = TodoList()

    while True:
        print("\n=== Todo List ===")
        print("1. Add task")
        print("2. List tasks")
        print("3. Complete task")
        print("4. Delete task")
        print("5. Show all (including completed)")
        print("6. Quit")

        choice = input("\nSelect: ").strip()

        if choice == "1":
            title = input("Task title: ")
            priority = input("Priority (low/medium/high): ") or "medium"
            todo.add(title, priority)
        elif choice == "2":
            todo.list_tasks()
        elif choice == "3":
            try:
                task_id = int(input("Task ID to complete: "))
                todo.complete(task_id)
            except ValueError:
                print("Invalid ID!")
        elif choice == "4":
            try:
                task_id = int(input("Task ID to delete: "))
                todo.delete(task_id)
            except ValueError:
                print("Invalid ID!")
        elif choice == "5":
            todo.list_tasks(show_all=True)
        elif choice == "6":
            print("Goodbye!")
            break
        else:
            print("Invalid choice!")

if __name__ == "__main__":
    main()
```

## Step-by-Step Explanation

1. **TodoList Class**: Encapsulates all todo functionality.
2. **Persistence**: JSON file stores tasks between sessions.
3. **CRUD Operations**: Create, Read, Update, Delete tasks.
4. **Task Structure**: Each task has ID, title, priority, status, and timestamp.

| Bug | Cause | Fix |
|-----|-------|-----|
| Tasks not saved | Missing save() call | Call save() after every modification |
| ID conflicts | Simple increment | Use UUID or track max ID |
| File not found | First run | Check file existence before loading |
