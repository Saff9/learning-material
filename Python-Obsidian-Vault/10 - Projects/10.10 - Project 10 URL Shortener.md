```python
# url_shortener.py
"""A simple URL shortener with file-based storage."""

import json
import os
import hashlib
import re
from datetime import datetime
from urllib.parse import urlparse

class URLShortener:
    def __init__(self, filename="urls.json"):
        self.filename = filename
        self.urls = {}
        self.load()

    def load(self):
        """Load URL mappings from file."""
        if os.path.exists(self.filename):
            with open(self.filename, 'r') as f:
                self.urls = json.load(f)

    def save(self):
        """Save URL mappings to file."""
        with open(self.filename, 'w') as f:
            json.dump(self.urls, f, indent=2)

    def validate_url(self, url):
        """Check if URL is valid."""
        pattern = r"^https?://[^\s/$.?#].[^\s]*$"
        return re.match(pattern, url) is not None

    def generate_code(self, url):
        """Generate a short code for URL."""
        # Use first 6 characters of MD5 hash
        hash_obj = hashlib.md5(url.encode())
        return hash_obj.hexdigest()[:6]

    def shorten(self, url):
        """Shorten a URL."""
        if not self.validate_url(url):
            print("Invalid URL! Must start with http:// or https://")
            return None

        # Check if already shortened
        for code, data in self.urls.items():
            if data["url"] == url:
                print(f"URL already shortened: {code}")
                return code

        code = self.generate_code(url)

        # Handle collisions
        while code in self.urls:
            code = hashlib.md5(f"{url}{datetime.now()}".encode()).hexdigest()[:6]

        self.urls[code] = {
            "url": url,
            "created": datetime.now().isoformat(),
            "clicks": 0
        }
        self.save()

        print(f"Shortened: {url}")
        print(f"Code: {code}")
        return code

    def expand(self, code):
        """Get original URL from short code."""
        if code in self.urls:
            self.urls[code]["clicks"] += 1
            self.save()
            return self.urls[code]["url"]
        return None

    def get_stats(self, code):
        """Get statistics for a short URL."""
        if code in self.urls:
            data = self.urls[code]
            print(f"\nURL: {data['url']}")
            print(f"Created: {data['created']}")
            print(f"Clicks: {data['clicks']}")
            return data
        print("Code not found!")
        return None

    def list_all(self):
        """List all shortened URLs."""
        if not self.urls:
            print("No URLs shortened yet!")
            return

        print("\n=== Shortened URLs ===")
        for code, data in self.urls.items():
            print(f"{code} -> {data['url']} (Clicks: {data['clicks']})")

def main():
    shortener = URLShortener()

    while True:
        print("\n=== URL Shortener ===")
        print("1. Shorten URL")
        print("2. Expand URL")
        print("3. View stats")
        print("4. List all")
        print("5. Quit")

        choice = input("\nSelect: ").strip()

        if choice == "1":
            url = input("Enter URL to shorten: ")
            shortener.shorten(url)

        elif choice == "2":
            code = input("Enter short code: ")
            original = shortener.expand(code)
            if original:
                print(f"Original URL: {original}")
            else:
                print("Code not found!")

        elif choice == "3":
            code = input("Enter short code: ")
            shortener.get_stats(code)

        elif choice == "4":
            shortener.list_all()

        elif choice == "5":
            print("Goodbye!")
            break

        else:
            print("Invalid choice!")

if __name__ == "__main__":
    main()
```

## Step-by-Step Explanation

1. **URL Validation**: Regex ensures valid HTTP/HTTPS URLs.
2. **Hash Generation**: MD5 hash creates unique short codes.
3. **Collision Handling**: Regenerates code if duplicate exists.
4. **Click Tracking**: Counts how many times each URL is accessed.
