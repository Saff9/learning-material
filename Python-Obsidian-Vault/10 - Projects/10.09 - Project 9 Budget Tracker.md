```python
# budget_tracker.py
"""A personal budget tracking application."""

import csv
import os
from datetime import datetime
from collections import defaultdict

class Transaction:
    def __init__(self, date, description, amount, category, type_="expense"):
        self.date = date
        self.description = description
        self.amount = float(amount)
        self.category = category
        self.type = type_  # "income" or "expense"

    def to_dict(self):
        return {
            "date": self.date,
            "description": self.description,
            "amount": self.amount,
            "category": self.category,
            "type": self.type
        }

class BudgetTracker:
    def __init__(self, filename="budget.csv"):
        self.filename = filename
        self.transactions = []
        self.load()

    def load(self):
        """Load transactions from CSV."""
        if os.path.exists(self.filename):
            with open(self.filename, 'r', newline='') as f:
                reader = csv.DictReader(f)
                for row in reader:
                    self.transactions.append(Transaction(**row))

    def save(self):
        """Save transactions to CSV."""
        with open(self.filename, 'w', newline='') as f:
            fieldnames = ["date", "description", "amount", "category", "type"]
            writer = csv.DictWriter(f, fieldnames=fieldnames)
            writer.writeheader()
            for t in self.transactions:
                writer.writerow(t.to_dict())

    def add_transaction(self, description, amount, category, type_="expense"):
        """Add a new transaction."""
        date = datetime.now().strftime("%Y-%m-%d")
        transaction = Transaction(date, description, amount, category, type_)
        self.transactions.append(transaction)
        self.save()
        print(f"Added: {description} - ${amount}")

    def get_balance(self):
        """Calculate current balance."""
        income = sum(t.amount for t in self.transactions if t.type == "income")
        expenses = sum(t.amount for t in self.transactions if t.type == "expense")
        return income - expenses

    def get_summary(self):
        """Get spending summary by category."""
        summary = defaultdict(lambda: {"income": 0, "expense": 0})

        for t in self.transactions:
            summary[t.category][t.type] += t.amount

        return summary

    def show_report(self):
        """Display financial report."""
        print("\n=== Budget Report ===")

        income = sum(t.amount for t in self.transactions if t.type == "income")
        expenses = sum(t.amount for t in self.transactions if t.type == "expense")
        balance = income - expenses

        print(f"Total Income:  ${income:.2f}")
        print(f"Total Expenses: ${expenses:.2f}")
        print(f"Balance:        ${balance:.2f}")

        print("\n--- By Category ---")
        summary = self.get_summary()
        for category, amounts in sorted(summary.items()):
            net = amounts["income"] - amounts["expense"]
            print(f"{category:15} Income: ${amounts['income']:.2f}  Expense: ${amounts['expense']:.2f}  Net: ${net:.2f}")

    def list_transactions(self, limit=10):
        """List recent transactions."""
        if not self.transactions:
            print("No transactions yet!")
            return

        print("\n=== Recent Transactions ===")
        for t in reversed(self.transactions[-limit:]):
            sign = "+" if t.type == "income" else "-"
            print(f"{t.date} | {t.description:20} | {sign}${t.amount:.2f} | {t.category}")

def main():
    tracker = BudgetTracker()

    while True:
        print("\n=== Budget Tracker ===")
        print(f"Current Balance: ${tracker.get_balance():.2f}")
        print("1. Add income")
        print("2. Add expense")
        print("3. View report")
        print("4. Recent transactions")
        print("5. Quit")

        choice = input("\nSelect: ").strip()

        if choice == "1":
            desc = input("Description: ")
            amount = float(input("Amount: "))
            category = input("Category: ")
            tracker.add_transaction(desc, amount, category, "income")

        elif choice == "2":
            desc = input("Description: ")
            amount = float(input("Amount: "))
            category = input("Category (food/transport/entertainment/bills/other): ")
            tracker.add_transaction(desc, amount, category, "expense")

        elif choice == "3":
            tracker.show_report()

        elif choice == "4":
            tracker.list_transactions()

        elif choice == "5":
            print("Goodbye!")
            break

        else:
            print("Invalid choice!")

if __name__ == "__main__":
    main()
```

## Step-by-Step Explanation

1. **Transaction Class**: Stores individual financial transactions.
2. **BudgetTracker Class**: Manages all transactions and calculations.
3. **CSV Persistence**: Stores data in a simple CSV format.
4. **Reporting**: Calculates totals by category and overall balance.
