```python
# web_scraper.py
"""A simple web scraper for extracting article titles."""

import requests
from bs4 import BeautifulSoup
import csv
import time
import urllib.robotparser

class WebScraper:
    def __init__(self, delay=1):
        self.delay = delay
        self.session = requests.Session()
        self.session.headers.update({
            "User-Agent": "Python Learning Scraper/1.0"
        })

    def check_robots(self, url):
        """Check if scraping is allowed."""
        rp = urllib.robotparser.RobotFileParser()
        rp.set_url(f"{url}/robots.txt")
        try:
            rp.read()
            return rp.can_fetch("*", url)
        except:
            return True  # If no robots.txt, assume allowed

    def fetch(self, url):
        """Fetch a webpage."""
        try:
            response = self.session.get(url, timeout=10)
            response.raise_for_status()
            return response.text
        except requests.RequestException as e:
            print(f"Error fetching {url}: {e}")
            return None

    def parse_articles(self, html, selector):
        """Extract articles from HTML."""
        soup = BeautifulSoup(html, "html.parser")
        articles = []

        for item in soup.select(selector):
            title = item.get_text(strip=True)
            link = item.get("href", "")
            articles.append({"title": title, "link": link})

        return articles

    def scrape(self, url, selector, output_file=None):
        """Scrape and optionally save data."""
        print(f"Scraping: {url}")

        if not self.check_robots(url):
            print("Scraping not allowed by robots.txt")
            return []

        time.sleep(self.delay)  # Be polite

        html = self.fetch(url)
        if not html:
            return []

        articles = self.parse_articles(html, selector)

        if output_file:
            self.save_to_csv(articles, output_file)

        return articles

    def save_to_csv(self, data, filename):
        """Save data to CSV file."""
        if not data:
            return

        with open(filename, 'w', newline='', encoding='utf-8') as f:
            writer = csv.DictWriter(f, fieldnames=data[0].keys())
            writer.writeheader()
            writer.writerows(data)

        print(f"Saved {len(data)} items to {filename}")

def main():
    scraper = WebScraper(delay=2)

    # Example: Scrape a blog (replace with actual URL)
    url = "https://example-blog.com"
    selector = "h2.article-title a"

    articles = scraper.scrape(url, selector, "articles.csv")

    print(f"\nFound {len(articles)} articles:")
    for article in articles[:5]:
        print(f"- {article['title']}")

if __name__ == "__main__":
    main()
```

## Step-by-Step Explanation

1. **Session Setup**: Reuses connections with custom headers.
2. **Robots Check**: Respects website scraping policies.
3. **Polite Delay**: Waits between requests to avoid overwhelming servers.
4. **Data Extraction**: Uses CSS selectors to find specific elements.
