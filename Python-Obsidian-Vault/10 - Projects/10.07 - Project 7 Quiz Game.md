```python
# quiz_game.py
"""An interactive quiz game with multiple categories."""

import json
import random
import time
from pathlib import Path

class QuizGame:
    def __init__(self, questions_file="questions.json"):
        self.questions = self.load_questions(questions_file)
        self.score = 0
        self.total_questions = 0
        self.streak = 0
        self.best_streak = 0

    def load_questions(self, filename):
        """Load questions from JSON file."""
        default_questions = {
            "python": [
                {
                    "question": "What is the output of 2 ** 3?",
                    "options": ["6", "8", "9", "4"],
                    "answer": 1
                },
                {
                    "question": "Which function converts a string to integer?",
                    "options": ["str()", "int()", "float()", "bool()"],
                    "answer": 1
                },
                {
                    "question": "What does `len()` do?",
                    "options": ["Returns length", "Returns type", "Returns id", "Returns hash"],
                    "answer": 0
                }
            ],
            "general": [
                {
                    "question": "What is the capital of France?",
                    "options": ["London", "Berlin", "Paris", "Madrid"],
                    "answer": 2
                },
                {
                    "question": "Which planet is known as the Red Planet?",
                    "options": ["Venus", "Mars", "Jupiter", "Saturn"],
                    "answer": 1
                }
            ]
        }

        if Path(filename).exists():
            with open(filename, 'r') as f:
                return json.load(f)
        return default_questions

    def ask_question(self, question_data):
        """Ask a single question."""
        print(f"\n{question_data['question']}")
        for i, option in enumerate(question_data["options"]):
            print(f"{i + 1}. {option}")

        start_time = time.time()

        while True:
            try:
                answer = input("\nYour answer (1-4): ").strip()
                if answer.lower() == 'q':
                    return None

                answer = int(answer) - 1
                if 0 <= answer < len(question_data["options"]):
                    elapsed = time.time() - start_time
                    return answer, elapsed
                else:
                    print("Invalid choice. Enter 1-4.")
            except ValueError:
                print("Please enter a number.")

    def run_quiz(self, category=None, num_questions=5):
        """Run the quiz."""
        print("\n=== Python Quiz Game ===")
        print("Answer the questions. Enter 'q' to quit.")

        # Select questions
        if category and category in self.questions:
            available = self.questions[category]
        else:
            available = []
            for cat_questions in self.questions.values():
                available.extend(cat_questions)

        selected = random.sample(available, min(num_questions, len(available)))

        for i, q in enumerate(selected, 1):
            print(f"\n--- Question {i} of {len(selected)} ---")
            print(f"Score: {self.score} | Streak: {self.streak}")

            result = self.ask_question(q)
            if result is None:
                break

            answer, elapsed = result
            self.total_questions += 1

            if answer == q["answer"]:
                points = max(10 - int(elapsed), 1)
                self.score += points
                self.streak += 1
                self.best_streak = max(self.best_streak, self.streak)
                print(f"✓ Correct! +{points} points (Time: {elapsed:.1f}s)")
            else:
                self.streak = 0
                correct = q["options"][q["answer"]]
                print(f"✗ Wrong! The answer was: {correct}")

        self.show_results()

    def show_results(self):
        """Display final results."""
        print("\n=== Quiz Complete ===")
        print(f"Questions answered: {self.total_questions}")
        print(f"Final score: {self.score}")
        print(f"Best streak: {self.best_streak}")

        if self.total_questions > 0:
            accuracy = (self.score / (self.total_questions * 10)) * 100
            print(f"Accuracy: {accuracy:.1f}%")

def main():
    game = QuizGame()

    print("Categories:")
    print("1. Python")
    print("2. General Knowledge")
    print("3. Mixed")

    choice = input("\nSelect category (1-3): ").strip()

    categories = {"1": "python", "2": "general", "3": None}
    category = categories.get(choice)

    game.run_quiz(category)

if __name__ == "__main__":
    main()
```

## Step-by-Step Explanation

1. **Question Loading**: JSON file stores questions with options and answers.
2. **Timer**: Tracks response time for scoring.
3. **Streak System**: Rewards consecutive correct answers.
4. **Categories**: Supports multiple quiz categories.
