## Full Source Code

```python
# calculator.py
"""A command-line calculator with basic operations."""

def add(a, b):
    """Add two numbers."""
    return a + b

def subtract(a, b):
    """Subtract b from a."""
    return a - b

def multiply(a, b):
    """Multiply two numbers."""
    return a * b

def divide(a, b):
    """Divide a by b."""
    if b == 0:
        raise ValueError("Cannot divide by zero")
    return a / b

def power(a, b):
    """Raise a to the power of b."""
    return a ** b

def get_number(prompt):
    """Get a valid number from user."""
    while True:
        try:
            return float(input(prompt))
        except ValueError:
            print("Invalid input. Please enter a number.")

def display_menu():
    """Display the calculator menu."""
    print("\n=== Python Calculator ===")
    print("1. Add")
    print("2. Subtract")
    print("3. Multiply")
    print("4. Divide")
    print("5. Power")
    print("6. Quit")

def main():
    """Main calculator loop."""
    operations = {
        "1": ("Add", add),
        "2": ("Subtract", subtract),
        "3": ("Multiply", multiply),
        "4": ("Divide", divide),
        "5": ("Power", power)
    }

    while True:
        display_menu()
        choice = input("\nSelect operation (1-6): ").strip()

        if choice == "6":
            print("Goodbye!")
            break

        if choice not in operations:
            print("Invalid choice. Please try again.")
            continue

        name, func = operations[choice]
        print(f"\n{name}")

        num1 = get_number("Enter first number: ")
        num2 = get_number("Enter second number: ")

        try:
            result = func(num1, num2)
            print(f"\nResult: {result}")
        except ValueError as e:
            print(f"\nError: {e}")
        except Exception as e:
            print(f"\nUnexpected error: {e}")

if __name__ == "__main__":
    main()
```

## Step-by-Step Explanation

1. **Function Definitions**: Each operation is a separate function for modularity.
2. **Input Validation**: `get_number()` ensures valid numeric input.
3. **Menu System**: Dictionary maps choices to operations.
4. **Error Handling**: Try-except catches division by zero and other errors.
5. **Main Loop**: Continuously runs until user chooses to quit.
