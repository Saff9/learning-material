```python
# file_organizer.py
"""Organizes files in a directory by their extensions."""

import os
import shutil
from pathlib import Path
from collections import defaultdict

class FileOrganizer:
    def __init__(self, source_dir):
        self.source_dir = Path(source_dir)
        self.stats = defaultdict(int)

    def organize_by_extension(self):
        """Organize files into folders by extension."""
        if not self.source_dir.exists():
            print(f"Directory {self.source_dir} does not exist!")
            return

        for file_path in self.source_dir.iterdir():
            if file_path.is_file():
                ext = file_path.suffix.lower()
                if not ext:
                    ext = "no_extension"
                else:
                    ext = ext[1:]  # Remove the dot

                target_dir = self.source_dir / ext
                target_dir.mkdir(exist_ok=True)

                target_path = target_dir / file_path.name

                # Handle duplicates
                counter = 1
                while target_path.exists():
                    stem = file_path.stem
                    suffix = file_path.suffix
                    target_path = target_dir / f"{stem}_{counter}{suffix}"
                    counter += 1

                shutil.move(str(file_path), str(target_path))
                self.stats[ext] += 1
                print(f"Moved: {file_path.name} -> {ext}/")

        self.print_stats()

    def organize_by_date(self):
        """Organize files by modification date."""
        for file_path in self.source_dir.iterdir():
            if file_path.is_file():
                mtime = file_path.stat().st_mtime
                date_folder = Path(time.strftime("%Y-%m", time.localtime(mtime)))

                target_dir = self.source_dir / date_folder
                target_dir.mkdir(exist_ok=True)

                shutil.move(str(file_path), str(target_dir / file_path.name))
                print(f"Moved: {file_path.name} -> {date_folder}/")

    def clean_empty_folders(self):
        """Remove empty folders."""
        removed = 0
        for dir_path in self.source_dir.rglob("*"):
            if dir_path.is_dir() and not any(dir_path.iterdir()):
                dir_path.rmdir()
                removed += 1
        print(f"Removed {removed} empty folders")

    def print_stats(self):
        """Print organization statistics."""
        print("\n=== Organization Complete ===")
        total = sum(self.stats.values())
        print(f"Total files organized: {total}")
        for ext, count in sorted(self.stats.items()):
            print(f"  .{ext}: {count} files")

def main():
    import time

    print("=== File Organizer ===")
    directory = input("Enter directory to organize: ").strip()

    if not directory:
        directory = "."

    print("\n1. Organize by extension")
    print("2. Organize by date")
    print("3. Clean empty folders")

    choice = input("\nSelect: ").strip()

    organizer = FileOrganizer(directory)

    if choice == "1":
        organizer.organize_by_extension()
    elif choice == "2":
        organizer.organize_by_date()
    elif choice == "3":
        organizer.clean_empty_folders()
    else:
        print("Invalid choice!")

if __name__ == "__main__":
    main()
```

## Step-by-Step Explanation

1. **Pathlib**: Modern path handling with Path objects.
2. **Extension Detection**: Uses `suffix` property to get file extension.
3. **Duplicate Handling**: Renames files if target already exists.
4. **Statistics**: Tracks how many files were moved to each folder.
