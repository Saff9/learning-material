## The `else` Clause

```python
try:
    number = int(input("Enter a number: "))
except ValueError:
    print("That is not a number!")
else:
    # Runs ONLY if no exception occurred
    print(f"You entered: {number}")
    result = number * 2
    print(f"Double: {result}")
```

## The `finally` Clause

```python
try:
    file = open("data.txt", "r")
    content = file.read()
except FileNotFoundError:
    print("File not found!")
finally:
    # ALWAYS runs, even if exception occurred
    print("Cleaning up...")
    file.close()  # Make sure file is closed
```

## Complete Structure

```python
try:
    print("Trying...")
    result = 10 / 2
except ZeroDivisionError:
    print("Division by zero!")
else:
    print(f"Success! Result: {result}")
finally:
    print("This always runs")

# Output:
# Trying...
# Success! Result: 5.0
# This always runs
```

## With Exception

```python
try:
    print("Trying...")
    result = 10 / 0
except ZeroDivisionError:
    print("Division by zero!")
else:
    print(f"Success! Result: {result}")
finally:
    print("This always runs")

# Output:
# Trying...
# Division by zero!
# This always runs
```

## Practical Pattern: Resource Management

```python
def process_data(filename):
    file = None
    try:
        file = open(filename, "r")
        data = file.read()
        return int(data) * 2
    except (FileNotFoundError, ValueError) as e:
        print(f"Error: {e}")
        return None
    finally:
        if file:
            file.close()
            print("File closed")
```

## Using with Statement (Better Alternative)

```python
# The with statement is preferred for resources
def process_data(filename):
    try:
        with open(filename, "r") as file:
            data = file.read()
            return int(data) * 2
    except (FileNotFoundError, ValueError) as e:
        print(f"Error: {e}")
        return None
```

### Exercise 1: Database Connection Pattern
```python
def query_database(query):
    connection = None
    try:
        connection = create_connection()
        result = connection.execute(query)
        return result
    except ConnectionError:
        print("Failed to connect")
        return None
    else:
        print("Query executed successfully")
    finally:
        if connection:
            connection.close()
            print("Connection closed")
```

### Exercise 2: Calculator with Cleanup
```python
def calculator():
    while True:
        try:
            a = float(input("First number (or 'q' to quit): "))
            b = float(input("Second number: "))
            op = input("Operation (+, -, *, /): ")
        except ValueError:
            print("Invalid input. Try again.")
            continue

        try:
            if op == "+":
                result = a + b
            elif op == "-":
                result = a - b
            elif op == "*":
                result = a * b
            elif op == "/":
                result = a / b
            else:
                print("Unknown operation")
                continue
        except ZeroDivisionError:
            print("Cannot divide by zero!")
        else:
            print(f"Result: {result}")
        finally:
            print("---")
```
