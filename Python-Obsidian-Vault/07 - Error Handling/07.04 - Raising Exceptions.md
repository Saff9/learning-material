## Raising Built-in Exceptions

```python
def divide(a, b):
    if b == 0:
        raise ZeroDivisionError("Cannot divide by zero")
    return a / b

def get_item(items, index):
    if index < 0 or index >= len(items):
        raise IndexError(f"Index {index} out of range")
    return items[index]

def parse_age(age_string):
    age = int(age_string)
    if age < 0:
        raise ValueError("Age cannot be negative")
    return age
```

## Raising Custom Exceptions

```python
class ValidationError(Exception):
    pass

def validate_username(username):
    if len(username) < 3:
        raise ValidationError("Username must be at least 3 characters")
    if not username.isalnum():
        raise ValidationError("Username must be alphanumeric")
    return True
```

## Re-raising Exceptions

```python
def process_file(filename):
    try:
        with open(filename, "r") as f:
            return f.read()
    except FileNotFoundError:
        print(f"Warning: {filename} not found")
        raise  # Re-raise the same exception
```

## Exception Chaining

```python
def read_config(filename):
    try:
        with open(filename, "r") as f:
            return json.load(f)
    except FileNotFoundError as e:
        raise ConfigError(f"Config file missing: {filename}") from e
    except json.JSONDecodeError as e:
        raise ConfigError(f"Invalid JSON in {filename}") from e
```

## From None (Suppress Context)

```python
try:
    int("not a number")
except ValueError:
    raise CustomError("Invalid input") from None
```

## Assertions

```python
def calculate_discount(price, discount):
    assert price > 0, "Price must be positive"
    assert 0 <= discount <= 1, "Discount must be between 0 and 1"
    return price * (1 - discount)

# Note: Assertions are removed when Python runs with -O flag
```

## 🧪 Practice

### Exercise 1: Input Validator
```python
def get_positive_number(prompt):
    value = input(prompt)
    try:
        num = float(value)
        if num <= 0:
            raise ValueError("Number must be positive")
        return num
    except ValueError as e:
        raise ValueError(f"Invalid input: {value}. {e}") from e

try:
    result = get_positive_number("Enter a positive number: ")
    print(f"You entered: {result}")
except ValueError as e:
    print(e)
```

### Exercise 2: API Error Handler
```python
class APIError(Exception):
    pass

class NotFoundError(APIError):
    pass

class UnauthorizedError(APIError):
    pass

def api_call(endpoint):
    # Simulated API response
    responses = {
        "/users": {"status": 200, "data": []},
        "/admin": {"status": 401, "error": "Unauthorized"},
        "/missing": {"status": 404, "error": "Not found"}
    }

    response = responses.get(endpoint, {"status": 500})

    if response["status"] == 404:
        raise NotFoundError(f"Endpoint {endpoint} not found")
    elif response["status"] == 401:
        raise UnauthorizedError("Authentication required")
    elif response["status"] != 200:
        raise APIError(f"API error: {response}")

    return response["data"]

try:
    data = api_call("/admin")
except UnauthorizedError as e:
    print(f"Please log in: {e}")
```
